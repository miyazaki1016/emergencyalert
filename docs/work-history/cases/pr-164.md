+"""Read-only observation of EXACTLY two Vercel projects; never controls deployments.
+
+Deployment history is NOT authoritative quota usage. This program cannot enable a gate.
+"""
+import datetime as dt
+import json
+import os
+import sys
+import urllib.error
+import urllib.parse
+import urllib.request
+
+BASE = os.getenv("VERCEL_API_BASE", "https://api.vercel.com").rstrip("/")
+TOKEN = os.getenv("VERCEL_TOKEN")
+EXPECTED_TEAM = "team_VIfp8TXktY4DegRNpPhFb5uT"
+EXPECTED_PROJECTS = {
+    "prj_71NgSxGkNXkKECXc9jJRrJHdAkDS": "misaki",
+    "prj_LDqpPAqNGDaY4Aaix8ulwPc8oXmu": "emergencyalert",
+}
+TEAM = os.getenv("VERCEL_TEAM_ID")
+PROJECTS = {p.strip() for p in os.getenv("VERCEL_PROJECT_IDS", "").split(",") if p.strip()}
+PAGE_LIMIT = 100
+MAX_PAGES = 100
+
+
+def request_json(path, params=None):
+    url = BASE + path
+    if params:
+        url += "?" + urllib.parse.urlencode(params)
+    req = urllib.request.Request(
+        url, headers={"Authorization": "Bearer " + TOKEN, "Accept": "application/json"}
+    )
+    with urllib.request.urlopen(req, timeout=20) as response:
+        data = json.load(response)
+    if not isinstance(data, dict) or data.get("error"):
+        raise RuntimeError("invalid Vercel API response")
+    return data
+
+
+def fetch(params):
+    return request_json("/v6/deployments", params)
+
+
+def verify_projects():
+    """Verify project ID, name, and team individually: no default first-page assumptions."""
+    for project_id, name in EXPECTED_PROJECTS.items():
+        data = request_json("/v9/projects/" + project_id)
+        if (data.get("id") != project_id or data.get("name") != name
+                or data.get("accountId") != EXPECTED_TEAM):
+            raise RuntimeError("project membership not verified: " + project_id)
+
+
+def record_time(record):
+    value = record.get("created")
+    if value is None:
+        value = record.get("createdAt")
+    if isinstance(value, bool) or not isinstance(value, (int, float)) or value < 0:
+        raise RuntimeError("deployment record has invalid created timestamp")
+    return int(value)

+

+    """Request each project separately and reject ambiguous/truncated pages."""
+    seen = set()
+    deployments = []
+    for project_id in EXPECTED_PROJECTS:

+        for _ in range(MAX_PAGES):

+                raise RuntimeError("invalid deployments response; monitoring incomplete")
+            if not items:
+                break
+            times = []
+            for item in items:
+                if not isinstance(item, dict):
+                    raise RuntimeError("invalid deployment item")
+                key = item.get("uid") or item.get("id")
+                if not isinstance(key, str) or not key:
+                    raise RuntimeError("missing deployment identifier")
+                created = record_time(item)
+                times.append(created)
+                # The API can include other projects despite a request filter.
+                if item.get("projectId") != project_id:
+                    raise RuntimeError("API returned foreign/unknown project")
+                if before is not None and created > before:
+                    raise RuntimeError("pagination boundary violated")
+                if key not in seen and created >= cutoff:

+                    deployments.append(item)

+            next_before = oldest - 1
+            if before is not None and next_before >= before:
+                raise RuntimeError("pagination did not advance")
+            before = next_before
+        else:
+            raise RuntimeError("pagination safety limit reached; count incomplete")
+    return deployments

+        return "unknown"
+    ratio = usage / limit
+    if ratio >= .95:
+        return "critical"
+    if ratio >= .85:
+        return "high"
+    if ratio >= .70:
+        return "warning"
+    return "normal"

+    now = dt.datetime.now(dt.timezone.utc)
+    cutoff = int((now - dt.timedelta(hours=24)).timestamp() * 1000)
+    hour = int((now - dt.timedelta(hours=1)).timestamp() * 1000)

+    print(json.dumps({
+        "team_id": EXPECTED_TEAM,
+        "projects_filter": sorted(EXPECTED_PROJECTS),
+        "observed_at": now.isoformat(),
+        "window": "rolling",
+        "deployment_records_24h": len(deployments),

+        "status": "unknown",
+        "status_basis": "authoritative_quota_unavailable",
+        "quota_alert": quota_alert(None, 100),
+        "safe_to_gate": False,
+        "quota_verified": False,
+        "note": "History only; NOT quota used/remaining. No automatic pause/resume.",
+    }, ensure_ascii=False))
+    return 0
+

+    try:
+        sys.exit(main())
+    except (urllib.error.URLError, ValueError, RuntimeError, KeyError) as exc:
+        print(json.dumps({"status": "unknown", "error": str(exc), "safe_to_gate": False}), file=sys.stderr)
+        sys.exit(1)

+        In particular, an earlier snapshot becoming stale after insertion is
+        possible; this journal is not a production deployment gate.

+            return Decision("HOLD", "invalid simulation calendar day")
+        try:
+            if date.fromisoformat(day).isoformat() != day:
+                return Decision("HOLD", "invalid simulation calendar day")
+        except ValueError:
+            return Decision("HOLD", "invalid simulation calendar day")
+        if not isinstance(request, dict) or not isinstance(github_snapshot, dict):
+            return Decision("HOLD", "missing mock inputs")
+        if not isinstance(request.get("project_id"), str) or request["project_id"] not in ALLOWED_PROJECTS:
+            return Decision("HOLD", "project not in allowlist")
+        if not isinstance(request.get("request_id"), str) or not request["request_id"].strip():
+            return Decision("HOLD", "invalid request identity")
+        with self._connect() as db:
+            db.execute("BEGIN IMMEDIATE")
+            try:
+                seen = {row[0] for row in db.execute("SELECT request_id FROM requests")}
+                counts = db.execute(
+                    "SELECT emergency, COUNT(*) FROM requests WHERE day=? GROUP BY emergency",
+                    (day,),
+                ).fetchall()
+                per_kind = dict(counts)
+                # Existing caller-supplied data is never a source of authority
+                # for journal counts or request identities.
+                mock = dict(github_snapshot)
+                mock["seen_request_ids"] = seen
+                mock["normal_requests_today"] = per_kind.get(0, 0)
+                mock["emergency_requests_today"] = per_kind.get(1, 0)
+                decision = evaluate(request, mock)
+                if decision.status != "REVIEW_READY":
+                    db.execute("ROLLBACK")
+                    return decision
+                db.execute(
+                    "INSERT INTO requests (request_id, project_id, sha, day, emergency) "
+                    "VALUES (?, ?, ?, ?, ?)",
+                    (request["request_id"], request["project_id"], request["sha"],
+                     day, int(request["emergency"])),
+                )
+                db.execute("COMMIT")
+                return Decision("RECORDED_FOR_MANUAL_REVIEW",
+                                "offline journal only; no deployment authority")
+            except BaseException:
+                db.execute("ROLLBACK")
+                raise

+        with self._connect() as db:
+            rows = db.execute(
+                "SELECT emergency, COUNT(*) FROM requests WHERE day=? GROUP BY emergency",
+                (day,),
+            ).fetchall()

+"""Offline-only Vercel release request decision model.
+
+IMPORTANT: This module has no network, GitHub writes, or deployment capability.
+REVIEW_READY is a request for a human to review, NEVER permission to deploy.
+Internal request budgets are voluntary workflow controls, NOT Vercel quota.
+"""
+from dataclasses import dataclass
+import re
+
+ALLOWED_PROJECTS = frozenset({
+    "prj_71NgSxGkNXkKECXc9jJRrJHdAkDS",
+    "prj_LDqpPAqNGDaY4Aaix8ulwPc8oXmu",
+})
+NORMAL_DAILY_REQUEST_BUDGET = 5
+EMERGENCY_DAILY_REQUEST_BUDGET = 2
+SHA = re.compile(r"^[0-9a-f]{40}$")
+
+
+@dataclass(frozen=True)
+class Decision:
+    status: str
+    reason: str
+    can_auto_deploy: bool = False
+
+
+def evaluate(request, snapshot):
+    """Fail-closed, pure triage. Never authorizes Vercel deployment.
+
+    request: project_id, sha, request_id, requester, approved_by, emergency
+    snapshot: trusted, latest_sha_by_project, seen_request_ids, ci_green_shas,
+              normal_requests_today, emergency_requests_today
+
+    The caller MUST obtain the latest GitHub SHA, checks and approvals anew;
+    the function cannot authenticate any caller-supplied facts.
+    GitHub-side locking/persistence is required before enabling any gate.
+    """
+    def hold(reason):
+        return Decision("HOLD", reason)
+
+    if not isinstance(request, dict) or not isinstance(snapshot, dict):
+        return hold("missing request or trusted snapshot")
+    if snapshot.get("trusted") is not True:
+        return hold("snapshot unavailable or untrusted")
+    project = request.get("project_id")
+    sha = request.get("sha")
+    rid = request.get("request_id")
+    requester = request.get("requester")
+    approver = request.get("approved_by")
+    emergency = request.get("emergency")
+    if not isinstance(project, str) or project not in ALLOWED_PROJECTS:
+        return hold("project outside two-project allowlist")
+    if not isinstance(sha, str) or not SHA.fullmatch(sha):
+        return hold("invalid pinned commit SHA")
+    if not isinstance(rid, str) or not rid.strip():
+        return hold("missing idempotency key")
+    if not isinstance(requester, str) or not requester:
+        return hold("missing requester")
+    if not isinstance(emergency, bool):
+        return hold("invalid emergency flag")
+    latest = snapshot.get("latest_sha_by_project")
+    seen = snapshot.get("seen_request_ids")
+    green = snapshot.get("ci_green_shas")
+    if not isinstance(latest, dict) or not isinstance(seen, (list, tuple, set, frozenset)) or not isinstance(green, (list, tuple, set, frozenset)):
+        return hold("incomplete authoritative GitHub snapshot")
+    if not all(isinstance(x, str) for x in seen) or not all(isinstance(x, str) for x in green):
+        return hold("malformed GitHub snapshot")
+    # Enforce uniqueness BEFORE evaluating approvals and quotas.
+    if rid in seen:
+        return Decision("DUPLICATE", "request already recorded")
+    if latest.get(project) != sha:
+        return hold("stale commit; only latest approved SHA may be considered")
+    if sha not in green:
+        return hold("required GitHub checks are not green for pinned SHA")
+    if not isinstance(approver, str) or not approver or requester == approver:
+        return hold("independent human approval required")
+    normal = snapshot.get("normal_requests_today")
+    urgent = snapshot.get("emergency_requests_today")
+    if any(isinstance(x, bool) or not isinstance(x, int) or x < 0 for x in (normal, urgent)):
+        return hold("internal request ledger unavailable")
+    if emergency:
+        if urgent >= EMERGENCY_DAILY_REQUEST_BUDGET:
+            return hold("voluntary emergency request reserve exhausted")
+        if request.get("emergency_authorized") is not True:
+            return hold("explicit emergency authorization required")
+    elif normal >= NORMAL_DAILY_REQUEST_BUDGET:
+        return hold("voluntary normal request budget exhausted; emergency reserve untouched")
+    return Decision("REVIEW_READY", "manual review only; deployment disabled")

+"""Single-process ONLY simulation of atomic release-request intake.
+
+No GitHub/Vercel network calls, secrets, deployment, or production gate.
+The lock does NOT protect multiple GitHub Actions runners. An actual runner
+would require durable storage with atomic CAS and server-side approvals.
+"""
+from copy import deepcopy
+from threading import Lock
+
+try:
+    from scripts.vercel_release_request_policy import evaluate, Decision
+except ModuleNotFoundError:
+    from vercel_release_request_policy import evaluate, Decision
+
+
+class SimulationLedger:
+    """Model concurrent admissions and replacement of stale requests offline."""
+
+    def __init__(self, latest_sha_by_project, ci_green_shas):
+        self._lock = Lock()
+        self._seen = set()
+        self._pending = {}
+        self._latest = dict(latest_sha_by_project)
+        self._ci_green = set(ci_green_shas)
+        self._normal_count = 0
+        self._emergency_count = 0
+        self._revision = 0
+
+    def submit(self, request):
+        with self._lock:
+            snapshot = {
+                "trusted": True,  # Simulation fixture, NOT real GitHub attestation
+                "latest_sha_by_project": dict(self._latest),
+                "ci_green_shas": set(self._ci_green),
+                "seen_request_ids": set(self._seen),
+                "normal_requests_today": self._normal_count,
+                "emergency_requests_today": self._emergency_count,
+            }
+            decision = evaluate(request, snapshot)
+            if decision.status != "REVIEW_READY":
+                return decision
+            project = request["project_id"]
+            # Supersede any earlier request for the same project. The earlier
+            # SHA is never replayed, but audit identity is remembered.
+            previous = self._pending.get(project)
+            if previous:
+                self._seen.add(previous["request_id"])
+            self._pending[project] = deepcopy(request)
+            self._seen.add(request["request_id"])
+            if request["emergency"]:
+                self._emergency_count += 1
+            else:
+                self._normal_count += 1
+            self._revision += 1
+            return Decision("RECORDED_FOR_MANUAL_REVIEW", "no deployment authority")
+
+    def update_latest(self, project_id, sha, ci_green=False):
+        """Simulate a new commit; invalidate a superseded pending request."""
+        with self._lock:
+            if project_id not in self._latest or not isinstance(sha, str):
+                raise ValueError("unknown project or invalid SHA")
+            self._latest[project_id] = sha
+            if ci_green:
+                self._ci_green.add(sha)
+            if project_id in self._pending and self._pending[project_id]["sha"] != sha:
+                self._pending.pop(project_id)
+            self._revision += 1
+
+    def snapshot(self):
+        with self._lock:
+            return {
+                "revision": self._revision,
+                "pending": deepcopy(self._pending),
+                "seen_request_ids": sorted(self._seen),
+                "normal_requests_today": self._normal_count,
+                "emergency_requests_today": self._emergency_count,
+                "can_auto_deploy": False,
+            }
