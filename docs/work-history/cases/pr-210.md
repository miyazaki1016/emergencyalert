+"use client";

+import { useEffect, useState } from "react";
+import { forecastTime, type NationalForecastSnapshot } from "../../lib/weather/rain/nationalForecastSnapshot";
+
+const formatTime = new Intl.DateTimeFormat("ja-JP", {
+  timeZone: "Asia/Tokyo", month: "numeric", day: "numeric", hour: "2-digit", minute: "2-digit", hourCycle: "h23",
+});
+
+export default function NationalHeavyRainForecast() {
+  const [forecast, setForecast] = useState<NationalForecastSnapshot | null>();
+  useEffect(() => {
+    const controller = new AbortController();
+    async function refresh() {
+      try {
+        const response = await fetch("/api/rain/national-forecast", { signal: controller.signal });
+        if (!response.ok) throw new Error("Forecast unavailable");
+        const snapshot: NationalForecastSnapshot = await response.json();
+        if (!controller.signal.aborted) setForecast(snapshot);
+      } catch {
+        if (!controller.signal.aborted) setForecast(null);
+      }
+    }
+    void refresh();
+    const interval = window.setInterval(refresh, 60_000);
+    return () => {
+      controller.abort();
+      window.clearInterval(interval);
+    };
+  }, []);
+
+  const rows = forecast?.frames.flatMap(frame => frame.municipalities.map(area => ({
+    ...area, validTime: frame.validTime,
+  })));
+
+  return (
+    <section aria-labelledby="national-rain-heading">
+      <h2 id="national-rain-heading">全国大雨予報</h2>
+      <p>今後30分以内に、1時間あたり30mm以上の雨が予測される地域</p>
+      {forecast === undefined && <p role="status">全国の予報を取得中…</p>}
+      {forecast === null && <p role="status">現在、全国の予報を確認できません。</p>}
+      {forecast && (
+        <>
+          <p>気象庁 · {formatTime.format(forecastTime(forecast.basetime))}発表（日本時間）</p>
+          {rows?.length === 0 ? <p>取得できた予報の時間帯では、該当地域はありません。</p> : (
+            <ul style={{ listStyle: "none", padding: 0 }}>
+              {rows?.map(area => (
+                <li key={`${area.code}:${area.validTime}`} style={{ padding: "12px 0", borderBottom: "1px solid #ffffff33" }}>
+                  <strong>{area.prefecture} {area.municipality}</strong>
+                  <div>{formatTime.format(forecastTime(area.validTime))}ごろ · {area.minimumMmPerHour}mm/h以上</div>
+                </li>
+              ))}
+            </ul>
+          )}
+        </>
+      )}
+    </section>
+  );
+}

+const STRENGTHS = [30, 50, 80] as const;
+
+export type NationalForecastSnapshot = {
+  source: "JMA";
+  n03Version: string;
+  basetime: string;
+  frames: {
+    validTime: string;
+    municipalities: {
+      code: string;
+      prefecture: string;
+      municipality: string;
+      minimumMmPerHour: (typeof STRENGTHS)[number];
+    }[];
+  }[];
+};
+
+export function forecastTime(value: string) {
+  return Date.parse(value.replace(/^(\d{4})(\d{2})(\d{2})(\d{2})(\d{2})(\d{2})$/, "$1-$2-$3T$4:$5:$6Z"));
+}
+
+export function readNationalForecastSnapshot(value: unknown, now = Date.now()): NationalForecastSnapshot {
+  const snapshot = value as NationalForecastSnapshot;
+  if (snapshot?.source !== "JMA" || !/^\d{8}$/.test(snapshot.n03Version)
+    || !Number.isFinite(forecastTime(snapshot.basetime))
+    || !Array.isArray(snapshot.frames) || snapshot.frames.length !== 6) {
+    throw new Error("Invalid forecast snapshot");
+  }
+
+  const base = forecastTime(snapshot.basetime);
+  if (base > now || now - base > 15 * 60_000) throw new Error("Stale forecast snapshot");
+  let previous = 0;
+  for (const frame of snapshot.frames) {
+    const time = forecastTime(frame.validTime);
+    if (!Number.isFinite(time) || time <= base || (previous && time - previous !== 300_000)
+      || !Array.isArray(frame.municipalities) || !frame.municipalities.every(area =>
+        /^\d{5}$/.test(area?.code) && typeof area.prefecture === "string"
+        && typeof area.municipality === "string" && STRENGTHS.includes(area.minimumMmPerHour),
+      )) {
+      throw new Error("Invalid forecast frame");
+    }
+    previous = time;
+  }
+
+  const frames = snapshot.frames.filter(frame => {
+    const time = forecastTime(frame.validTime);
+    return time > now && time <= now + 30 * 60_000;
+  });
+  if (!frames.length) throw new Error("No current forecast frames");
+  return { ...snapshot, frames };
+}

+async function scanForecast(frame: JmaTargetTime, load: ReturnType<typeof localPreparedLoader>["load"]) {
+  const signal = AbortSignal.timeout(60_000);
+  async function fetchTile(zoom: number, x: number, y: number) {
+    const response = await fetch(buildJmaRainTileUrl(frame, zoom, x, y), { signal });
+    if (!response.ok) throw new Error(`JMA tile unavailable: ${frame.validtime} z${zoom}/${x}/${y} (${response.status})`);

+  }
+
+  const coarse = (await Promise.all(japanTiles.map(async ([x, y]) =>
+    scanHeavyRainTile(await fetchTile(4, x, y), 4, x, y, 1),
+  ))).flat();
+
+  // A full scan of each selected detail tile covers all candidate windows at
+  // once. Refine directly from z4 so intermediate levels cannot drop a window.
+  const tiles = new Map(stagedRefinementJobsFromCoarseCandidates(coarse, frame, 10).map(job =>
+    [`${job.tileX}:${job.tileY}`, job],
+  ));
+  const municipalities = new Map<string, {
+    code: string;
+    prefecture: string;
+    municipality: string;
+    minimumMmPerHour: 30 | 50 | 80;

+    const candidates = scanHeavyRainTile(await fetchTile(10, tileX, tileY), 10, tileX, tileY, 1);
+    for (const { level, minimumMmPerHour } of thresholds) {
+      const footprint = heavyRainAreaPolygons(candidates.filter(candidate => candidate.level === level), 10);
+      const areas = await resolveNationalRainMunicipalities(footprint, load, undefined, signal);
+      for (const area of areas) {
+        if ((municipalities.get(area.code)?.minimumMmPerHour ?? 0) < minimumMmPerHour) {
+          municipalities.set(area.code, { ...area, minimumMmPerHour });
+        }
+      }
+    }
+  }
+
+  return [...municipalities.values()].sort((a, b) => a.code.localeCompare(b.code));
+}

+  }
