```text
+function polygonBounds(polygon: number[][][]): Bounds {
+  let west = Infinity, south = Infinity, east = -Infinity, north = -Infinity;
+  for (const ring of polygon) for (const [x, y] of ring) {
+    west = Math.min(west, x); south = Math.min(south, y);
+    east = Math.max(east, x); north = Math.max(north, y);
+  }
+  return [west, south, east, north];
+}

+function boundsIntersect(a: Bounds, b: Bounds) {
+  return a[0] <= b[2] && a[2] >= b[0] && a[1] <= b[3] && a[3] >= b[1];
+}

+function pointInRing([x, y]: Position, ring: number[][]) {
+  let inside = false;
+  for (let i = 0, j = ring.length - 1; i < ring.length; j = i++) {
+    const [xi, yi] = ring[i];
+    const [xj, yj] = ring[j];
+    const crosses = yi > y !== yj > y && x < ((xj - xi) * (y - yi)) / (yj - yi) + xi;
+    if (crosses) inside = !inside;
+  }
+  return inside;
+}

+  return !polygon.slice(1).some((hole) => pointInRing(point, hole));
+}

+  return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0]);
+}

+function onSegment(a: Position, b: Position, p: Position) {
+  if (Math.abs(cross(a, b, p)) > EPSILON) return false;
+  return p[0] >= Math.min(a[0], b[0]) - EPSILON &&
+    p[0] <= Math.max(a[0], b[0]) + EPSILON &&
+    p[1] >= Math.min(a[1], b[1]) - EPSILON &&
+    p[1] <= Math.max(a[1], b[1]) + EPSILON;
+}

+  const abC = cross(a, b, c);
+  const abD = cross(a, b, d);
+  const cdA = cross(c, d, a);
+  const cdB = cross(c, d, b);
+
+  if (Math.abs(abC) <= EPSILON && onSegment(a, b, c)) return true;
+  if (Math.abs(abD) <= EPSILON && onSegment(a, b, d)) return true;
+  if (Math.abs(cdA) <= EPSILON && onSegment(c, d, a)) return true;
+  if (Math.abs(cdB) <= EPSILON && onSegment(c, d, b)) return true;
+
+  return (abC > EPSILON) !== (abD > EPSILON) &&

+}

+    for (let j = 1; j < b.length; j++) {
+      if (segmentsIntersect(a[i - 1] as Position, a[i] as Position, b[j - 1] as Position, b[j] as Position)) return true;
+    }

+    ? [area.geometry.coordinates as number[][][]]
+    : area.geometry.coordinates as number[][][][];
+  const rainBounds = polygonBounds(rainPolygon);
+  return areaPolygons.some((polygon) =>
+    boundsIntersect(rainBounds, polygonBounds(polygon)) && polygonsIntersect(rainPolygon, polygon),

+}

+// repeated forecast frames do not rescan every coastline vertex.

+  polygons: Array<{ polygon: number[][][]; bounds: Bounds }>;

+  const cached = preparedAreaCache.get(area);
+  if (cached) return cached;
+  const geometry = area.geometry.type === "Polygon"
+    ? [area.geometry.coordinates as number[][][]]
+    : area.geometry.coordinates as number[][][][];
+  const polygons = geometry.map((polygon) => ({ polygon, bounds: polygonBounds(polygon) }));
+  const prepared: PreparedArea = {
+    area, polygons,
+    west: Math.min(...polygons.map(({ bounds }) => bounds[0])),
+    south: Math.min(...polygons.map(({ bounds }) => bounds[1])),
+    east: Math.max(...polygons.map(({ bounds }) => bounds[2])),
+    north: Math.max(...polygons.map(({ bounds }) => bounds[3])),
+  };
+  preparedAreaCache.set(area, prepared);
+  return prepared;
+}

+  const rainWithBounds = rainPolygons.map((rain) => ({
+    rain,
+    bounds: polygonBounds(rain.coordinates),
+  }));
+  const areasWithBounds = areas.map(prepareArea);
+
+  return areasWithBounds
+    .filter(({ polygons, west, south, east, north }) =>
+      rainWithBounds.some(({ rain, bounds: rainBounds }) => {
+        if (west > rainBounds[2] || east < rainBounds[0] ||
+            south > rainBounds[3] || north < rainBounds[1]) return false;
+        return polygons.some(({ polygon, bounds: areaBounds }) =>
+          boundsIntersect(rainBounds, areaBounds) &&
+          polygonsIntersect(rain.coordinates, polygon),
+        );
+      }),
+    )
+    .map(({ area }) => area);
+}
```

```text
+// Cache exact pixel-footprint intersections across forecast frames. Values:
+// 0 = not evaluated, 1 = no intersection, 2 = intersects.
+// Bound memory to 64 MiB; cache keys include area identity and tile coordinates.
+const MAX_MASK_BYTES = 64 * 1024 * 1024;
+const PIXELS_PER_TILE = 256 * 256;
+const masks = new Map<string, { area: AdministrativeArea; values: Uint8Array }>();
+let maskBytes = 0;
+const areaIds = new WeakMap<AdministrativeArea, number>();
+let nextAreaId = 0;
+function areaId(area: AdministrativeArea): number {
+  let id = areaIds.get(area);
+  if (id === undefined) { id = ++nextAreaId; areaIds.set(area, id); }
+  return id;
+}
+function pixelMask(area: AdministrativeArea, zoom: number, tileX: number, tileY: number): Uint8Array {
+  const key = `${areaId(area)}:${zoom}:${tileX}:${tileY}`;
+  const existing = masks.get(key);
+  if (existing && existing.area === area) {
+    masks.delete(key);
+    masks.set(key, existing);
+    return existing.values;
+  }
+  while (maskBytes + PIXELS_PER_TILE > MAX_MASK_BYTES) {
+    const oldest = masks.keys().next().value;
+    if (oldest === undefined) break;
+    masks.delete(oldest);
+    maskBytes -= PIXELS_PER_TILE;
+  }
+  const values = new Uint8Array(PIXELS_PER_TILE);
+  masks.set(key, { area, values });
+  maskBytes += PIXELS_PER_TILE;
+  return values;
+}
+
+/** Proof-only N03 municipality intersection; every rainy pixel's entire
+ * footprint is tested, including edges and holes. No stride or center shortcut.
+ * Cached decisions are exact results from the existing geometry matcher.
+ */
+export function municipalitiesForLightRainTile(
+  tile: Buffer, zoom: number, tileX: number, tileY: number, areas: AdministrativeArea[],
+): Array<Pick<AdministrativeArea, "code" | "prefecture" | "municipality">> {
+  const pixels = locateJmaRainPixelsAtLeast1mm(tile, zoom, tileX, tileY);
+  if (!pixels.length || !areas.length) return [];
+  const size = 256 * 2 ** zoom;
+  const longitude = (x: number) => x / size * 360 - 180;
+  const latitude = (y: number) => Math.atan(Math.sinh(Math.PI - 2 * Math.PI * y / size)) * 180 / Math.PI;
+  const footprint = new Map<number, HeavyRainPolygon>();
+  function rainFootprint(pixelX: number, pixelY: number): HeavyRainPolygon {
+    const index = pixelY * 256 + pixelX;
+    const cached = footprint.get(index);
+    if (cached) return cached;
+    const x = tileX * 256 + pixelX, y = tileY * 256 + pixelY;
+    const w = longitude(x), e = longitude(x + 1), n = latitude(y), s = latitude(y + 1);
+    const polygon: HeavyRainPolygon = { coordinates: [[[w, n], [e, n], [e, s], [w, s], [w, n]]] };
+    footprint.set(index, polygon);
+    return polygon;
+  }
+  const matched: Array<Pick<AdministrativeArea, "code" | "prefecture" | "municipality">> = [];
+  for (const area of areas) {
+    const mask = pixelMask(area, zoom, tileX, tileY);
+    let hit = false;
+    for (const { pixelX, pixelY } of pixels) {
+      const index = pixelY * 256 + pixelX;
+      let decision = mask[index];
+      if (decision === 0) {
+        decision = affectedAdministrativeAreas([rainFootprint(pixelX, pixelY)], [area]).length ? 2 : 1;
+        mask[index] = decision;
+      }
+      if (decision === 2) { hit = true; break; }
+    }
+    if (hit) matched.push({ code: area.code, prefecture: area.prefecture, municipality: area.municipality });
+  }
+  return matched;
```

```text
+import { createHash } from "node:crypto";
+import { readFileSync } from "node:fs";
+import { join } from "node:path";
+import {
+  PIXEL_MASK_ALGORITHM_VERSION, PIXEL_MASK_FORMAT_VERSION,
+  decodeTilePixelMasks, type TilePixelMasks,
+} from "./lightRainPixelMasks";
+
+type Manifest = {
+  format: string; formatVersion: number; algorithmVersion: number;
+  n03Version: string; zoom: number; prefectures: string[];
+  municipalities: number; tiles: number;
+  files: Record<string, { bytes: number; sha256: string }>;
+};
+const ALL_PREFECTURES = Array.from({ length: 47 }, (_, i) => String(i + 1).padStart(2, "0"));
+export function openNationwidePixelMasks(root: string, n03Version: string) {
+  const manifest = JSON.parse(readFileSync(join(root, "manifest.json"), "utf8")) as Manifest;
+  if (manifest.format !== "light-rain-pixel-masks" ||
+      manifest.formatVersion !== PIXEL_MASK_FORMAT_VERSION ||
+      manifest.algorithmVersion !== PIXEL_MASK_ALGORITHM_VERSION ||
+      manifest.n03Version !== n03Version || manifest.zoom !== 10 ||
+      !Array.isArray(manifest.prefectures) ||
+      ALL_PREFECTURES.some(p => !manifest.prefectures.includes(p)) ||
+      !manifest.files || Object.keys(manifest.files).length !== manifest.tiles ||
+      !Number.isInteger(manifest.municipalities) || manifest.municipalities <= 0) {
+    throw new Error("PIXEL_MASK_MANIFEST_INCOMPATIBLE");
+  }
+  // The generated nationwide manifest is only ~100KB; never retain all
+  // 35.9MiB of masks in memory. Read a tile on demand.
+  return {
+    has(tileX: number, tileY: number): boolean {
+      return Object.hasOwn(manifest.files, `z10/${tileX}/${tileY}.bin`);
+    },
+    load(tileX: number, tileY: number): TilePixelMasks {
+      const relative = `z10/${tileX}/${tileY}.bin`;
+      const meta = manifest.files[relative];
+      if (!meta) throw new Error(`PIXEL_MASK_MISSING: ${relative}`);
+      const bytes = readFileSync(join(root, relative));
+      if (bytes.length !== meta.bytes ||
+          createHash("sha256").update(bytes).digest("hex") !== meta.sha256) {
+        throw new Error(`PIXEL_MASK_CHECKSUM_MISMATCH: ${relative}`);
+      }
+      const masks = decodeTilePixelMasks(bytes);
+      if (masks.zoom !== 10 || masks.tileX !== tileX || masks.tileY !== tileY) {
+        throw new Error(`PIXEL_MASK_TILE_MISMATCH: ${relative}`);
+      }
+      return masks;
+    },
+  };
+}
```

```text
+/**
+ * Precomputed z10 tile x N03 administrative-area pixel correspondence masks.
+ *
+ * Independent deliverable for Issue #220. The nationwide scanner's exact
+ * matcher (`municipalitiesForLightRainTile`) spends most of its time testing
+ * rainy pixel footprints against coastline polygons, and the geometry never
+ * changes while the N03 dataset version is fixed. This module precomputes,
+ * for every pixel of a tile, whether that pixel's footprint intersects each
+ * administrative area, using the production matcher itself
+ * (`affectedAdministrativeAreas`) as the only decision function. Decisions
+ * are therefore identical by construction: no center-point approximation, no
+ * stride thinning, and boundary contact counts exactly as the production
+ * geometry code decides.
+ *
+ * Binary format (all integers little-endian):
+ *   bytes 0-3   magic "LRPM"
+ *   byte  4     format version (1)
+ *   byte  5     zoom
+ *   bytes 6-7   reserved (0)
+ *   bytes 8-11  tileX (uint32)
+ *   bytes 12-15 tileY (uint32)
+ *   bytes 16-19 entry count (uint32)
+ *   per entry:  uint16 code byte length, code bytes (UTF-8),
+ *               8192-byte pixel bitmask
+ * Bit order: pixel index p = pixelY * 256 + pixelX lives at byte p >> 3,
+ * bit p & 7 (least significant bit first). A set bit means the production
+ * matcher judges that pixel's footprint to intersect the area.
+ * Entries with an all-zero mask are never stored: an area missing from a
+ * tile file intersects no pixel of that tile.
+ */
+import {
+  affectedAdministrativeAreas,
+  type AdministrativeArea,
+  type HeavyRainPolygon,
+} from "./lightRainAdministrativeAreas";
+import { locateJmaRainPixelsAtLeast1mm } from "./lightRainProof";
+
+export const PIXEL_MASK_FORMAT_VERSION = 1;
+/** Bump when the decision semantics or footprint geometry change. */
+export const PIXEL_MASK_ALGORITHM_VERSION = 1;
+export const TILE_PIXELS = 256;
+export const PIXELS_PER_TILE = TILE_PIXELS * TILE_PIXELS;
+export const MASK_BYTES = PIXELS_PER_TILE / 8;
+
+const MAGIC = "LRPM";
+const HEADER_BYTES = 20;
+
+export interface TilePixelMaskEntry {
+  code: string;
+  mask: Uint8Array;
+}
+
+export interface TilePixelMasks {
+  zoom: number;
+  tileX: number;
+  tileY: number;
+  entries: TilePixelMaskEntry[];
+}
+
+/** Pixel footprint polygon, built with the same expressions as
+ * `municipalitiesForLightRainTile` so generated masks describe exactly the
+ * geometry the production matcher tests. */
+export function pixelFootprint(
+  zoom: number, tileX: number, tileY: number, pixelX: number, pixelY: number,
+): HeavyRainPolygon {
+  const size = TILE_PIXELS * 2 ** zoom;
+  const longitude = (x: number) => x / size * 360 - 180;
+  const latitude = (y: number) => Math.atan(Math.sinh(Math.PI - 2 * Math.PI * y / size)) * 180 / Math.PI;
+  const x = tileX * TILE_PIXELS + pixelX;
+  const y = tileY * TILE_PIXELS + pixelY;
+  const w = longitude(x), e = longitude(x + 1), n = latitude(y), s = latitude(y + 1);
+  return { coordinates: [[[w, n], [e, n], [e, s], [w, s], [w, n]]] };
+}
+
+type Bounds = [west: number, south: number, east: number, north: number];
+
+function polygonBounds(polygon: number[][][]): Bounds {
+  let west = Infinity, south = Infinity, east = -Infinity, north = -Infinity;
+  for (const ring of polygon) for (const [x, y] of ring) {
+    west = Math.min(west, x); south = Math.min(south, y);
+    east = Math.max(east, x); north = Math.max(north, y);
+  }
+  return [west, south, east, north];
+}
+
+/**
+ * Compute the full 256x256 intersection bitmask of one area over one tile by
+ * asking the production matcher about individual pixel footprints.
+ *
+ * Only pixels whose footprint bounds intersect one of the area's polygon
+ * bounds are evaluated. Every other pixel is decided "no intersection",
+ * which is exactly what the production matcher returns for it, because
+ * `affectedAdministrativeAreas` applies the same bounds test before its
+ * exact intersection test. The evaluated window is widened by one pixel on
+ * each side so floating-point inversion can never shrink it.
+ */
+export function computeAreaPixelMask(
+  zoom: number, tileX: number, tileY: number, area: AdministrativeArea,
+): Uint8Array {
+  const mask = new Uint8Array(MASK_BYTES);
+  const size = TILE_PIXELS * 2 ** zoom;
+  const polygons = area.geometry.type === "Polygon"
+    ? [area.geometry.coordinates as number[][][]]
+    : area.geometry.coordinates as number[][][][];
+  const tileWest = tileX * TILE_PIXELS, tileNorth = tileY * TILE_PIXELS;
+
+  for (const polygon of polygons) {
+    const [west, south, east, north] = polygonBounds(polygon);
+    // Global pixel window whose footprints can intersect these bounds.
+    const gx0 = Math.max(tileWest, Math.floor((west + 180) / 360 * size) - 1);
+    const gx1 = Math.min(tileWest + TILE_PIXELS - 1, Math.floor((east + 180) / 360 * size) + 1);
+    const latToGy = (lat: number) => {
+      const rad = Math.max(-85.05112878, Math.min(85.05112878, lat)) * Math.PI / 180;
+      return Math.floor((1 - Math.asinh(Math.tan(rad)) / Math.PI) / 2 * size);
+    };
+    const gy0 = Math.max(tileNorth, latToGy(north) - 1);
+    const gy1 = Math.min(tileNorth + TILE_PIXELS - 1, latToGy(south) + 1);
+    if (gx1 < gx0 || gy1 < gy0) continue;
+    for (let gy = gy0; gy <= gy1; gy++) {
+      for (let gx = gx0; gx <= gx1; gx++) {
+        const pixelX = gx - tileWest, pixelY = gy - tileNorth;
+        const index = pixelY * TILE_PIXELS + pixelX;
+        if (mask[index >> 3] & (1 << (index & 7))) continue;
+        const hit = affectedAdministrativeAreas(
+          [pixelFootprint(zoom, tileX, tileY, pixelX, pixelY)], [area],
+        ).length > 0;
+        if (hit) mask[index >> 3] |= 1 << (index & 7);
+      }
+    }
+  }
+  return mask;
+}
+
+export function maskHasPixel(mask: Uint8Array, pixelIndex: number): boolean {
+  return (mask[pixelIndex >> 3] & (1 << (pixelIndex & 7))) !== 0;
+}
+
+// --- Fast exact path --------------------------------------------------------
+// `computeAreaPixelMask` calls the production matcher once per pixel, which
+// is exact but slow on complex coastlines (a single tile can take minutes).
+// The functions below compute the identical mask with two accelerations
+// whose results are provably the same decisions, not approximations:
+//
+// 1. Segment grid: production `polygonsIntersect` first tests the footprint's
+//    4 edges against every outer-ring segment. A segment can only intersect
+//    the footprint if its bounding box overlaps the footprint's bounding box,
+//    so each segment is registered in every tile cell its bounding box
+//    overlaps (window widened by one pixel) and only those candidates run
+//    the production segment-intersection test, verbatim.
+// 2. Corner classification: production then tests whether the footprint's
+//    first vertex [west, north] lies in the area polygon with a ray-casting
+//    point-in-ring test. The footprint's first vertex is the pixel's
+//    north-west grid corner. For a fixed corner row (fixed y), every edge's
+//    ray-crossing x is a constant, so all 257x257 corner classifications of
+//    a ring are computed row by row with the identical crossing formula and
+//    floating-point operations, then combined exactly as `pointInPolygon`
+//    combines them (inside the outer ring and outside every hole ring).
+// 3. The remaining test, the area ring's first vertex inside the footprint,
+//    runs the production point-in-ring loop against the 5-point footprint
+//    ring unchanged.
+//
+// The parity script `scripts/light-rain-pixel-mask-parity.ts` verifies this
+// path against `computeAreaPixelMask` on real N03 data, pixel for pixel.
+
+const FAST_EPSILON = 1e-12;
+type Pos = [number, number];
+
+function fastCross(a: Pos, b: Pos, c: Pos): number {
+  return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0]);
+}
+
+function fastOnSegment(a: Pos, b: Pos, p: Pos): boolean {
+  if (Math.abs(fastCross(a, b, p)) > FAST_EPSILON) return false;
+  return p[0] >= Math.min(a[0], b[0]) - FAST_EPSILON &&
+    p[0] <= Math.max(a[0], b[0]) + FAST_EPSILON &&
+    p[1] >= Math.min(a[1], b[1]) - FAST_EPSILON &&
+    p[1] <= Math.max(a[1], b[1]) + FAST_EPSILON;
+}
+
+function fastSegmentsIntersect(a: Pos, b: Pos, c: Pos, d: Pos): boolean {
+  const abC = fastCross(a, b, c);
+  const abD = fastCross(a, b, d);
+  const cdA = fastCross(c, d, a);
+  const cdB = fastCross(c, d, b);
+  if (Math.abs(abC) <= FAST_EPSILON && fastOnSegment(a, b, c)) return true;
+  if (Math.abs(abD) <= FAST_EPSILON && fastOnSegment(a, b, d)) return true;
+  if (Math.abs(cdA) <= FAST_EPSILON && fastOnSegment(c, d, a)) return true;
+  if (Math.abs(cdB) <= FAST_EPSILON && fastOnSegment(c, d, b)) return true;
+  return (abC > FAST_EPSILON) !== (abD > FAST_EPSILON) &&
+    (cdA > FAST_EPSILON) !== (cdB > FAST_EPSILON);
+}
+
+function fastPointInRing([x, y]: Pos, ring: number[][]): boolean {
+  let inside = false;
+  for (let i = 0, j = ring.length - 1; i < ring.length; j = i++) {
+    const [xi, yi] = ring[i];
+    const [xj, yj] = ring[j];
+    const crosses = yi > y !== yj > y && x < ((xj - xi) * (y - yi)) / (yj - yi) + xi;
+    if (crosses) inside = !inside;
+  }
+  return inside;
+}
+
+const CORNERS = TILE_PIXELS + 1;
+
+/** Classify every tile grid corner against one ring with the production
+ * ray-casting predicate, hoisting each edge's crossing x per corner row. */
+function classifyRingCorners(
+  ring: number[][], cornerLons: Float64Array, cornerLats: Float64Array,
+): Uint8Array {
+  const result = new Uint8Array(CORNERS * CORNERS);
+  const crossings: number[] = [];
+  for (let cy = 0; cy < CORNERS; cy++) {
+    const y = cornerLats[cy];
+    crossings.length = 0;
+    for (let i = 0, j = ring.length - 1; i < ring.length; j = i++) {
+      const [xi, yi] = ring[i];
+      const [xj, yj] = ring[j];
+      if (yi > y !== yj > y) {
+        crossings.push(((xj - xi) * (y - yi)) / (yj - yi) + xi);
+      }
+    }
+    if (!crossings.length) continue;
+    crossings.sort((a, b) => a - b);
+    // Sweep corners from east to west: inside iff an odd number of
+    // crossings have crossing-x greater than the corner's x.
+    let remaining = crossings.length;
+    let inside = false;
+    for (let cx = CORNERS - 1; cx >= 0; cx--) {
+      const x = cornerLons[cx];
+      while (remaining > 0 && crossings[remaining - 1] > x) {
+        inside = !inside;
+        remaining--;
+      }
+      if (inside) result[cy * CORNERS + cx] = 1;
+    }
+  }
+  return result;
+}
+
+export function computeAreaPixelMaskFast(
+  zoom: number, tileX: number, tileY: number, area: AdministrativeArea,
+): Uint8Array {
+  const mask = new Uint8Array(MASK_BYTES);
+  const size = TILE_PIXELS * 2 ** zoom;
+  const longitude = (x: number) => x / size * 360 - 180;
+  const latitude = (y: number) => Math.atan(Math.sinh(Math.PI - 2 * Math.PI * y / size)) * 180 / Math.PI;
+  const tileWest = tileX * TILE_PIXELS, tileNorth = tileY * TILE_PIXELS;
+  const cornerLons = new Float64Array(CORNERS);
+  const cornerLats = new Float64Array(CORNERS);
+  for (let i = 0; i < CORNERS; i++) {
+    cornerLons[i] = longitude(tileWest + i);
+    cornerLats[i] = latitude(tileNorth + i);
+  }
+  const latToGy = (lat: number) => {
+    const rad = Math.max(-85.05112878, Math.min(85.05112878, lat)) * Math.PI / 180;
+    return Math.floor((1 - Math.asinh(Math.tan(rad)) / Math.PI) / 2 * size);
+  };
+  const polygons = area.geometry.type === "Polygon"
+    ? [area.geometry.coordinates as number[][][]]
+    : area.geometry.coordinates as number[][][][];
+
+  for (const polygon of polygons) {
+    const outer = polygon[0];
+    if (!outer || outer.length === 0) continue;
+    const [west, south, east, north] = polygonBounds(polygon);
+    const gx0 = Math.max(tileWest, Math.floor((west + 180) / 360 * size) - 1);
+    const gx1 = Math.min(tileWest + TILE_PIXELS - 1, Math.floor((east + 180) / 360 * size) + 1);
+    const gy0 = Math.max(tileNorth, latToGy(north) - 1);
+    const gy1 = Math.min(tileNorth + TILE_PIXELS - 1, latToGy(south) + 1);
+    if (gx1 < gx0 || gy1 < gy0) continue;
+
+    // Corner classification for pointInPolygon(footprint NW corner, polygon).
+    const inPoly = classifyRingCorners(outer, cornerLons, cornerLats);
+    for (const hole of polygon.slice(1)) {
+      const inHole = classifyRingCorners(hole, cornerLons, cornerLats);
+      for (let i = 0; i < inPoly.length; i++) if (inHole[i]) inPoly[i] = 0;
+    }
+
+    // Segment grid over tile cells (outer ring only, as in production).
+    const segAx: number[] = [], segAy: number[] = [], segBx: number[] = [], segBy: number[] = [];
+    const grid = new Map<number, number[]>();
+    for (let k = 1; k < outer.length; k++) {
+      const a = outer[k - 1], b = outer[k];
+      const segIndex = segAx.length;
+      segAx.push(a[0]); segAy.push(a[1]); segBx.push(b[0]); segBy.push(b[1]);
+      const sx0 = Math.max(tileWest, Math.floor((Math.min(a[0], b[0]) + 180) / 360 * size) - 1);
+      const sx1 = Math.min(tileWest + TILE_PIXELS - 1, Math.floor((Math.max(a[0], b[0]) + 180) / 360 * size) + 1);
+      const sy0 = Math.max(tileNorth, latToGy(Math.max(a[1], b[1])) - 1);
+      const sy1 = Math.min(tileNorth + TILE_PIXELS - 1, latToGy(Math.min(a[1], b[1])) + 1);
+      for (let gy = sy0; gy <= sy1; gy++) {
+        for (let gx = sx0; gx <= sx1; gx++) {
+          const cell = (gy - tileNorth) * TILE_PIXELS + (gx - tileWest);
+          const list = grid.get(cell);
+          if (list) list.push(segIndex); else grid.set(cell, [segIndex]);
+        }
+      }
+    }
+
+    const areaVertex = outer[0] as Pos;
+    for (let gy = gy0; gy <= gy1; gy++) {
+      for (let gx = gx0; gx <= gx1; gx++) {
+        const pixelX = gx - tileWest, pixelY = gy - tileNorth;
+        const index = pixelY * TILE_PIXELS + pixelX;
+        if (mask[index >> 3] & (1 << (index & 7))) continue;
+        const w = longitude(gx), e = longitude(gx + 1);
+        const n = latitude(gy), s = latitude(gy + 1);
+        let hit = false;
+        const candidates = grid.get(index);
+        if (candidates) {
+          const f: Pos[] = [[w, n], [e, n], [e, s], [w, s], [w, n]];
+          outer: for (const segIndex of candidates) {
+            const c: Pos = [segAx[segIndex], segAy[segIndex]];
+            const d: Pos = [segBx[segIndex], segBy[segIndex]];
+            for (let k = 1; k < f.length; k++) {
+              if (fastSegmentsIntersect(f[k - 1], f[k], c, d)) { hit = true; break outer; }
+            }
+          }
+        }
+        if (!hit && inPoly[pixelY * CORNERS + pixelX]) hit = true;
+        if (!hit && fastPointInRing(areaVertex, [[w, n], [e, n], [e, s], [w, s], [w, n]])) hit = true;
+        if (hit) mask[index >> 3] |= 1 << (index & 7);
+      }
+    }
+  }
+  return mask;
+}

+export function encodeTilePixelMasks(tile: TilePixelMasks): Buffer {
+  const entries = tile.entries.filter((entry) => entry.mask.some((byte) => byte !== 0));
+  let total = HEADER_BYTES;
+  const codeBytes = entries.map((entry) => Buffer.from(entry.code, "utf8"));
+  for (const bytes of codeBytes) total += 2 + bytes.length + MASK_BYTES;
+  const out = Buffer.alloc(total);
+  out.write(MAGIC, 0, "ascii");
+  out.writeUInt8(PIXEL_MASK_FORMAT_VERSION, 4);
+  out.writeUInt8(tile.zoom, 5);
+  out.writeUInt16LE(0, 6);
+  out.writeUInt32LE(tile.tileX, 8);
+  out.writeUInt32LE(tile.tileY, 12);
+  out.writeUInt32LE(entries.length, 16);
+  let offset = HEADER_BYTES;
+  entries.forEach((entry, i) => {
+    out.writeUInt16LE(codeBytes[i].length, offset);
+    offset += 2;
+    codeBytes[i].copy(out, offset);
+    offset += codeBytes[i].length;
+    out.set(entry.mask, offset);
+    offset += MASK_BYTES;
+  });
+  return out;
+}
+
+/** Strict decoder. Any inconsistency throws, so a corrupt or stale artifact
+ * fails closed instead of silently producing wrong municipalities. */
+export function decodeTilePixelMasks(buffer: Buffer): TilePixelMasks {
+  if (buffer.length < HEADER_BYTES || buffer.toString("ascii", 0, 4) !== MAGIC) {
+    throw new Error("PIXEL_MASK_BAD_MAGIC");
+  }
+  if (buffer.readUInt8(4) !== PIXEL_MASK_FORMAT_VERSION) {
+    throw new Error("PIXEL_MASK_UNSUPPORTED_FORMAT_VERSION");
+  }
+  const zoom = buffer.readUInt8(5);
+  const tileX = buffer.readUInt32LE(8);
+  const tileY = buffer.readUInt32LE(12);
+  const count = buffer.readUInt32LE(16);
+  const entries: TilePixelMaskEntry[] = [];
+  let offset = HEADER_BYTES;
+  for (let i = 0; i < count; i++) {
+    if (offset + 2 > buffer.length) throw new Error("PIXEL_MASK_TRUNCATED");
+    const codeLength = buffer.readUInt16LE(offset);
+    offset += 2;
+    if (offset + codeLength + MASK_BYTES > buffer.length) throw new Error("PIXEL_MASK_TRUNCATED");
+    const code = buffer.toString("utf8", offset, offset + codeLength);
+    offset += codeLength;
+    const mask = new Uint8Array(MASK_BYTES);
+    mask.set(buffer.subarray(offset, offset + MASK_BYTES));
+    offset += MASK_BYTES;
+    entries.push({ code, mask });
+  }
+  if (offset !== buffer.length) throw new Error("PIXEL_MASK_TRAILING_BYTES");
+  return { zoom, tileX, tileY, entries };
+}
+
+/**
+ * Drop-in counterpart of `municipalitiesForLightRainTile` driven by
+ * precomputed masks. Rain pixel extraction is shared with the production
+ * matcher, and the result order follows the `areas` argument exactly.
+ * Throws when no mask set is supplied for the tile, so the caller decides
+ * the fallback (the exact matcher) instead of receiving a silent empty
+ * result for a tile the artifact does not cover.
+ */
+export function municipalitiesForLightRainTileWithMasks(
+  tile: Buffer, zoom: number, tileX: number, tileY: number,
+  tileMasks: TilePixelMasks | undefined, areas: AdministrativeArea[],
+): Array<Pick<AdministrativeArea, "code" | "prefecture" | "municipality">> {
+  return matchRainPixelsToMasks(locateJmaRainPixelsAtLeast1mm(tile, zoom, tileX, tileY),
+    zoom, tileX, tileY, tileMasks, areas);
+}
+
+export function matchRainPixelsToMasks(
+  pixels: ReturnType<typeof locateJmaRainPixelsAtLeast1mm>,
+  zoom: number, tileX: number, tileY: number,
+  tileMasks: TilePixelMasks | undefined, areas: AdministrativeArea[],
+): Array<Pick<AdministrativeArea, "code" | "prefecture" | "municipality">> {
+  if (!pixels.length || !areas.length) return [];
+  if (!tileMasks || tileMasks.zoom !== zoom || tileMasks.tileX !== tileX || tileMasks.tileY !== tileY) {
+    throw new Error(`PIXEL_MASK_MISSING: z${zoom}/${tileX}/${tileY}`);
+  }
+  const byCode = new Map(tileMasks.entries.map((entry) => [entry.code, entry.mask]));
+  const matched: Array<Pick<AdministrativeArea, "code" | "prefecture" | "municipality">> = [];
+  for (const area of areas) {
+    const mask = byCode.get(area.code);
+    if (!mask) continue;
+    for (const { pixelX, pixelY } of pixels) {
+      if (maskHasPixel(mask, pixelY * TILE_PIXELS + pixelX)) {
+        matched.push({ code: area.code, prefecture: area.prefecture, municipality: area.municipality });
+        break;
+      }
+    }
+  }
+  return matched;
+}
+
+const HEAVY_RAIN_CLASSES = [
+  ["GTE_80", 80],
+  ["50_TO_80", 50],
+  ["30_TO_50", 30],
+] as const;
+
+export function matchHeavyRainPixelsToMasks(
+  pixels: ReturnType<typeof locateJmaRainPixelsAtLeast1mm>,
+  tileX: number, tileY: number, masks: TilePixelMasks, areas: AdministrativeArea[],
+) {
+  const matches = new Map<string, Pick<AdministrativeArea, "code" | "prefecture" | "municipality">
+    & { minimumMmPerHour: 30 | 50 | 80 }>();
+  for (const [intensity, minimumMmPerHour] of HEAVY_RAIN_CLASSES) {
+    for (const area of matchRainPixelsToMasks(
+      pixels.filter(pixel => pixel.intensityClass === intensity), 10, tileX, tileY, masks, areas,
+    )) {
+      if (!matches.has(area.code)) matches.set(area.code, { ...area, minimumMmPerHour });
+    }
+  }
+  return [...matches.values()];
+}
```

```text
+  const points = [];
+  const size = 256 * 2 ** zoom;
+  for (let py = 0; py < png.height; py++) for (let px = 0; px < png.width; px++) {
+    const i = (py * png.width + px) * 4;
+    const result = classifyRainPixel({ r: png.data[i], g: png.data[i + 1], b: png.data[i + 2], a: png.data[i + 3] });
+    if (result.status === "UNKNOWN_PIXEL") throw new Error("Unrecognized JMA pixel");

+    points.push({ pixelX: px, pixelY: py, intensityClass: result.intensityClass, longitude: worldX / size * 360 - 180,

+  }
```

```text
+    }

+    if (!existsSync(join(root, "manifest.json"))) {
+      const dirs = readdirSync(root, { withFileTypes: true }).filter(e => e.isDirectory());
+      if (dirs.length !== 1) throw new Error("LRPM_ARCHIVE_LAYOUT_UNEXPECTED");
+      root = join(root, dirs[0].name);
+    }
+    validate(root);
+    mkdirSync(resolve(destination!), { recursive: true });
+    // Never overwrite an existing version: fail rather than replace in-use masks.
+    if (existsSync(target)) throw new Error("LRPM_CACHE_ALREADY_EXISTS");
+    renameSync(root, target);
+    console.log(JSON.stringify({ event: "LRPM_CACHE_PREPARED", target, tiles: 733, sha256: digest }));

+  }
```

```text
## 合格待ちは担当AIが継続する（利用者の追加合図は不要）

「進めて」「りょ」等で依頼範囲が確定したら、Issue登録 → 結果取得 → 原本確認 → 判断の一括記録 → 再照合待機 → 合格確認 → 許可された本題を、可能な限り同じ応答内で連続実行する。判断記録だけで応答を終えたり、「続けてよいか」を利用者に聞いたりしない。合格前の案件作業禁止は維持する。

通常チャットでは、まずIssue本文とbotコメントを取得する。未完了なら環境で利用可能な待機機能を使い、目安15〜30秒間隔で再取得する（短時間の連続API呼出しは避ける）。待機中は本文を変更しない。新しい実行が止まるため、結果取得のためだけにIssueを繰り返し編集しない。要確認候補の判断は原本を読んで一括記録する。合格確認後は追加の「りょ」を待たず、元の許可範囲で続ける。

通常チャットでもIssueのbotコメントを間隔を空けて再取得し、最新の依頼本文・判断記録に対応する結果を待つ。「起動した」「判断を書いた」「結果待ち」は合格ではない。応答を続けられない場合は「照合待ち・案件未着手」で止め、次の応答では結果取得から再開する。利用者にサイト確認を求めたり、待機の代わりに案件作業を進めたりしない。

## 再利用判断の記録（利用者の承認は不要）

通常チャットではIssue本文の `history-preflight` ブロックへ記録し、本文編集で再照合する（形式は後述）。CLIを使う場合は、原本を読んだ担当AIが次のJSON配列をローカルファイルへ保存し、同じコマンドに `--decisions /tmp/preflight-decisions.json` を追加して再実行する。`id` と `evidence` は今回の出力からコピーする。3種類の判断のいずれかと、具体的な理由が必要。短い「確認済み」だけでは通らない。

~~~json
[
  {
    "id": "pr-62",
    "decision": "reuse",
    "reason": "47雨域の見逃し実測を再利用し、同じ比較実験は行わず全画素案の未解決条件だけを調べる",
    "evidence": "https://github.com/miyazaki1016/emergencyalert/pull/62"
  }
]
~~~

`reuse`＝既存成果を再利用、`extend`＝既存成果に不足する条件だけを追加、`distinct`＝原本と比較して別案件と判断。自動処理は原本を読んだか・理由が正しいかの意味までは証明できない。担当AIが根拠と理由の責任を持つ。

## Issue／PRでの自動チェック

`Existing work preflight (read-only)` が、Issue作成・編集・再開、PR作成・編集・コミット更新時に依頼内容を自動照合する。PRは変更パスも自動で読む。根拠を確認する記録が足りなければチェックは失敗し、ActionsのSummaryと `existing-work-preflight` 成果物に候補を残す。IssueではActionsの一覧から結果を確認する。Issueへの結果返却処理を導入する場合は、既存出力を専用のbotコメントへ記録する。PRへの自動コメントは追加しない。

PR／Issue本文には次の形式で判断を残す。JSON内容自体は依頼文の照合から除く。依頼文を消して照合を通すことや、根拠を読まずに候補を別案件扱いすることは禁止。

~~~text
<!-- history-preflight
{"decisions":[{"id":"pr-62","decision":"reuse","reason":"47雨域の見逃し実測を再利用し再測定を省略する。全画素案の未解決条件だけを確認する","evidence":"https://github.com/miyazaki1016/emergencyalert/pull/62"}]}
-->
~~~

Checksの `preflight` → Details → Summary、またはActionsの `Existing work preflight (read-only)` → Summaryを開く。自由な作業文を手動照合する場合はActionsのRun workflowで `task` に入力する。PR作成後のチェックは二重確認であり、着手前の照合（通常チャットはIssue経由、実行環境がある場合はCLI）を代替しない。

ベース側の照合プログラムを実行し、PRのコード・Issue本文はデータとして扱う。読取り権限のみ、8分上限。既存の日次更新・検索画面・承認ボードの役割は維持する。ブランチ保護の設定は変更しないため、チェックを無視するマージまで物理的に禁止するものではない。

## Issueへの結果返却の設計と安全条件

既存の `preflight.json`／`preflight.md` 成果物を、別の `publish-issue-result` ジョブでIssueの専用botコメントへ返す。照合器や判断方式は再実装しない。結果はREVIEW_REQUIRED／REVIEW_RECORDED／NO_STRONG_MATCH／INCOMPLETEのまま記録し、実行URLとID・試行番号を付ける。原本確認待ちの失敗も結果として読める。公開確認中はPUBLISHING（着手不可）とし、依頼とコメントを再取得して確認できた場合だけ `publication_status=confirmed` にする。本文に判断を書いて再照合すると、同じbotコメントが新しい結果へ更新される。

- 書込み権限はIssueイベント専用の返却ジョブの短期 `GITHUB_TOKEN` の `issues: write` だけに限定する。照合ジョブは引き続き読取り専用。返却ジョブはmainのコードを使い、PRイベントでは起動しない。APIキーや本番認証の変更は不要。
- 依頼本文・history-preflightは書き換えない。起動イベントに `issue_comment` を追加しないため、結果コメントの作成・更新で再照合ループは発生しない。
- 依頼本文・判断記録の指紋が実行時から変わった場合は旧結果を書き込まない。公開と本文編集が競合した場合はSUPERSEDED（更新前・着手不可）にする。新しい実行・試行の結果を古い実行で上書きしない。人のコメントや別用途のbotコメントも上書きしない。
- REVIEW_REQUIRED、INCOMPLETE、SUPERSEDED、PUBLISHING、結果未取得・不一致では着手しない。REVIEW_RECORDEDでも新規変更のCI成功・案件完成を意味しない。NO_STRONG_MATCHでは通常のコード・台帳確認を続ける。

通常チャットからは `fetch_issue_comments` 相当、またはGitHub汎用取得の `GET /repos/OWNER/REPO/issues/番号/comments` を使う。結果からrun_idが分かるので、利用者がサイトへ行って探す必要はない。[終了時の共通ルールと成功事例](WORK_HISTORY_HANDOFF_VALIDATION.md)も参照。
```

```text
"""Disposable Postgres only. No Supabase, credentials, endpoints or Push calls."""
import concurrent.futures, subprocess
T='11111111-1111-4111-8111-111111111111';S='22222222-2222-4222-8222-222222222222'
def sql(q):
 return subprocess.check_output(['psql','-q','-X','-A','-t','-v','ON_ERROR_STOP=1','-c','set role service_role; '+q],text=True).strip()
def race(q):
 with concurrent.futures.ThreadPoolExecutor(max_workers=2) as ex:return list(ex.map(sql,[q,q]))
leases=race(f"select coalesce(public.claim_rain_watch_target('{T}')::text,'BUSY')")
assert leases.count('BUSY')==1,leases
lease=next(x for x in leases if x!='BUSY')
claims=race(f"select coalesce(public.claim_rain_push_attempt('{T}','{lease}','episode/APPROACH','{S}')::text,'SKIP')")
assert claims.count('SKIP')==1,claims
attempt=next(x for x in claims if x!='SKIP')
assert sql('select count(*) from rain_push_attempts')=='1'
# Transport accepted, history/state update could fail. A new lease cannot resend.
sql(f"select public.finish_rain_push_attempt('{attempt}','ACCEPTED')")
sql(f"update rain_watch_runtime set leased_until=now()-interval '1 second'")
new=sql(f"select public.claim_rain_watch_target('{T}')")
assert sql(f"select public.save_rain_watch_runtime('{T}','{lease}','{{\"stale\":true}}')")=='f'
assert sql(f"select coalesce(public.claim_rain_push_attempt('{T}','{new}','episode/APPROACH','{S}')::text,'SKIP')")=='SKIP'
# Crash/unknown intent: no automatic retry even after lease expiration.
unknown=sql(f"select public.claim_rain_push_attempt('{T}','{new}','episode/SEVERE_1','{S}')")
sql("update rain_watch_runtime set leased_until=now()-interval '1 second'")
latest=sql(f"select public.claim_rain_watch_target('{T}')")
assert sql(f"select coalesce(public.claim_rain_push_attempt('{T}','{latest}','episode/SEVERE_1','{S}')::text,'SKIP')")=='SKIP'
# Escalation and a genuinely new episode retain separate send rights.
assert sql(f"select public.claim_rain_push_attempt('{T}','{latest}','episode/SEVERE_2','{S}')")
assert sql(f"select public.claim_rain_push_attempt('{T}','{latest}','new-episode/APPROACH','{S}')")
assert sql("select has_table_privilege('authenticated','rain_push_attempts','SELECT')")=='f'
assert sql("select has_function_privilege('anon','public.claim_rain_watch_target(uuid)','EXECUTE')")=='f'
assert sql("select count(*) from rain_push_attempts where status='ACCEPTED'")=='1'
print('PASS: 2 independent Postgres connections, exclusive target lease and subscription intent; stale writer fenced; accepted/unknown/crash not resent; escalation/new episode allowed; private permissions')
```
