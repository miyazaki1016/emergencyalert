+/** Reused from nationwide rain proof PR #59; isolated geometry matcher. */
+export type HeavyRainPolygon = { coordinates: number[][][] };
+
+export interface AdministrativeArea {
+  code: string;
+  prefecture: string;
+  municipality: string;
+  geometry: {
+    type: "Polygon" | "MultiPolygon";
+    coordinates: number[][][] | number[][][][];
+  };
+}
+
+type Position = [number, number];
+type Bounds = [west: number, south: number, east: number, north: number];
+
+function polygonBounds(polygon: number[][][]): Bounds {
+  let west = Infinity, south = Infinity, east = -Infinity, north = -Infinity;
+  for (const ring of polygon) for (const [x, y] of ring) {
+    west = Math.min(west, x); south = Math.min(south, y);
+    east = Math.max(east, x); north = Math.max(north, y);
+  }
+  return [west, south, east, north];
+}
+
+function boundsIntersect(a: Bounds, b: Bounds) {
+  return a[0] <= b[2] && a[2] >= b[0] && a[1] <= b[3] && a[3] >= b[1];
+}
+
+function pointInRing([x, y]: Position, ring: number[][]) {
+  let inside = false;
+  for (let i = 0, j = ring.length - 1; i < ring.length; j = i++) {
+    const [xi, yi] = ring[i];
+    const [xj, yj] = ring[j];
+    const crosses = yi > y !== yj > y && x < ((xj - xi) * (y - yi)) / (yj - yi) + xi;
+    if (crosses) inside = !inside;
+  }
+  return inside;
+}
+
+function pointInPolygon(point: Position, polygon: number[][][]) {
+  if (!polygon[0] || !pointInRing(point, polygon[0])) return false;
+  return !polygon.slice(1).some((hole) => pointInRing(point, hole));
+}
+
+const EPSILON = 1e-12;
+
+function cross(a: Position, b: Position, c: Position) {
+  return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0]);
+}
+
+function onSegment(a: Position, b: Position, p: Position) {
+  if (Math.abs(cross(a, b, p)) > EPSILON) return false;
+  return p[0] >= Math.min(a[0], b[0]) - EPSILON &&
+    p[0] <= Math.max(a[0], b[0]) + EPSILON &&
+    p[1] >= Math.min(a[1], b[1]) - EPSILON &&
+    p[1] <= Math.max(a[1], b[1]) + EPSILON;
+}
+
+function segmentsIntersect(a: Position, b: Position, c: Position, d: Position) {
+  const abC = cross(a, b, c);
+  const abD = cross(a, b, d);
+  const cdA = cross(c, d, a);
+  const cdB = cross(c, d, b);
+
+  if (Math.abs(abC) <= EPSILON && onSegment(a, b, c)) return true;
+  if (Math.abs(abD) <= EPSILON && onSegment(a, b, d)) return true;
+  if (Math.abs(cdA) <= EPSILON && onSegment(c, d, a)) return true;
+  if (Math.abs(cdB) <= EPSILON && onSegment(c, d, b)) return true;
+
+  return (abC > EPSILON) !== (abD > EPSILON) &&
+    (cdA > EPSILON) !== (cdB > EPSILON);
+}
+
+function ringsIntersect(a: number[][], b: number[][]) {
+  for (let i = 1; i < a.length; i++) {
+    for (let j = 1; j < b.length; j++) {
+      if (segmentsIntersect(a[i - 1] as Position, a[i] as Position, b[j - 1] as Position, b[j] as Position)) return true;
+    }
+  }
+  return false;
+}
+
+function polygonsIntersect(a: number[][][], b: number[][][]) {
+  const ar = a[0], br = b[0];
+  if (!ar || !br) return false;
+  return ringsIntersect(ar, br) ||
+    pointInPolygon(ar[0] as Position, b) ||
+    pointInPolygon(br[0] as Position, a);
+}
+
+export function rainPolygonIntersectsAdministrativeArea(rain: HeavyRainPolygon, area: AdministrativeArea) {
+  const rainPolygon = rain.coordinates;
+  const areaPolygons = area.geometry.type === "Polygon"
+    ? [area.geometry.coordinates as number[][][]]
+    : area.geometry.coordinates as number[][][][];
+  const rainBounds = polygonBounds(rainPolygon);
+  return areaPolygons.some((polygon) =>
+    boundsIntersect(rainBounds, polygonBounds(polygon)) && polygonsIntersect(rainPolygon, polygon),
+  );
+}
+
+export function affectedAdministrativeAreas(rainPolygons: HeavyRainPolygon[], areas: AdministrativeArea[]) {
+  const rainWithBounds = rainPolygons.map((rain) => ({
+    rain,
+    bounds: polygonBounds(rain.coordinates),
+  }));
+  const areasWithBounds = areas.map((area) => {
+    const polygons = area.geometry.type === "Polygon"
+      ? [area.geometry.coordinates as number[][][]]
+      : area.geometry.coordinates as number[][][][];
+    const prepared = polygons.map((polygon) => ({
+      polygon,
+      bounds: polygonBounds(polygon),
+    }));
+    return {
+      area,
+      polygons: prepared,
+      west: Math.min(...prepared.map(({ bounds }) => bounds[0])),
+      south: Math.min(...prepared.map(({ bounds }) => bounds[1])),
+      east: Math.max(...prepared.map(({ bounds }) => bounds[2])),
+      north: Math.max(...prepared.map(({ bounds }) => bounds[3])),
+    };
+  });
+
+  return areasWithBounds
+    .filter(({ polygons, west, south, east, north }) =>
+      rainWithBounds.some(({ rain, bounds: rainBounds }) => {
+        if (west > rainBounds[2] || east < rainBounds[0] ||
+            south > rainBounds[3] || north < rainBounds[1]) return false;
+        return polygons.some(({ polygon, bounds: areaBounds }) =>
+          boundsIntersect(rainBounds, areaBounds) &&
+          polygonsIntersect(rain.coordinates, polygon),
+        );
+      }),
+    )
+    .map(({ area }) => area);
+}

+import { locateJmaRainPixelsAtLeast1mm } from "./lightRainProof";

+/** Proof-only N03 municipality intersection, preserving each rainy pixel's full footprint. */
+export function municipalitiesForLightRainTile(
+  tile: Buffer, zoom: number, tileX: number, tileY: number, areas: AdministrativeArea[],
+): Array<Pick<AdministrativeArea, "code" | "prefecture" | "municipality">> {
+  const pixels = locateJmaRainPixelsAtLeast1mm(tile, zoom, tileX, tileY);
+  const size = 256 * 2 ** zoom;
+  const longitude = (x: number) => x / size * 360 - 180;
+  const latitude = (y: number) => Math.atan(Math.sinh(Math.PI - 2 * Math.PI * y / size)) * 180 / Math.PI;
+  const footprint: HeavyRainPolygon[] = pixels.map(({ pixelX, pixelY }) => {
+    const x = tileX * 256 + pixelX, y = tileY * 256 + pixelY;
+    const w = longitude(x), e = longitude(x + 1), n = latitude(y), s = latitude(y + 1);
+    return { coordinates: [[[w, n], [e, n], [e, s], [w, s], [w, n]]] };
+  });
+  return affectedAdministrativeAreas(footprint, areas)
+    .map(({ code, prefecture, municipality }) => ({ code, prefecture, municipality }));
+}

+import { readFileSync } from "node:fs";
+import type { AdministrativeArea } from "./lightRainAdministrativeAreas";
+import { municipalitiesForLightRainTile } from "./lightRainMunicipalities";
+
+/** Offline/read-only proof using prepared N03 .areas.json from PR #59.
+ * Never downloads production Storage data or writes to it.
+ */
+export function loadPreparedN03ForProof(path: string, expectedPrefectureCode: string): AdministrativeArea[] {
+  if (!/^\d{2}$/.test(expectedPrefectureCode)) throw new Error("Invalid prefecture code");
+  const parsed: unknown = JSON.parse(readFileSync(path, "utf8"));
+  if (!Array.isArray(parsed) || parsed.length === 0) throw new Error("Missing N03 municipality data");
+  const areas: AdministrativeArea[] = parsed.map((item: unknown) => {
+    if (!item || typeof item !== "object") throw new Error("Invalid N03 entry");
+    const a = item as Partial<AdministrativeArea>;
+    if (typeof a.code !== "string" || !/^\d{5}$/.test(a.code) ||
+      !a.code.startsWith(expectedPrefectureCode) ||
+      typeof a.prefecture !== "string" || !a.prefecture ||
+      typeof a.municipality !== "string" || !a.municipality ||
+      !a.geometry || !["Polygon", "MultiPolygon"].includes(a.geometry.type) ||
+      !Array.isArray(a.geometry.coordinates)) throw new Error("Invalid N03 municipality identity/geometry");
+    return a as AdministrativeArea;
+  });
+  if (new Set(areas.map(a => a.code)).size !== areas.length) throw new Error("Duplicate N03 municipality code");
+  return areas;
+}
+
+export function proveLightRainMunicipalitiesFromLocalN03(
+  tile: Buffer, zoom: number, tileX: number, tileY: number, n03Path: string, prefectureCode: string,
+) {
+  return municipalitiesForLightRainTile(tile, zoom, tileX, tileY,
+    loadPreparedN03ForProof(n03Path, prefectureCode));
+}

+type N03Geometry = AdministrativeArea["geometry"];
+type N03Feature = { type: "Feature"; properties?: Record<string, unknown> | null; geometry: N03Geometry | null };
+export type N03FeatureCollection = { type: "FeatureCollection"; features: N03Feature[] };
+
+/** Reused PR #59 official N03 feature conversion, isolated from production. */
+export function parseOfficialN03ForLightRain(collection: N03FeatureCollection): AdministrativeArea[] {
+  if (collection.type !== "FeatureCollection" || !Array.isArray(collection.features)) throw new Error("Invalid N03 GeoJSON");
+  const merged = new Map<string, AdministrativeArea>();
+  for (const feature of collection.features) {
+    if (!feature.geometry || !["Polygon", "MultiPolygon"].includes(feature.geometry.type)) continue;
+    const p = feature.properties ?? {};
+    const prefecture = typeof p.N03_001 === "string" ? p.N03_001.trim() : "";
+    const municipality = typeof p.N03_004 === "string" ? p.N03_004.trim() : "";
+    const code = typeof p.N03_007 === "string" ? p.N03_007.trim() : "";
+    if (!prefecture || !municipality || !/^\d{5}$/.test(code)) continue;
+    const next: AdministrativeArea = { code, prefecture, municipality, geometry: feature.geometry };
+    const previous = merged.get(code);
+    if (!previous) { merged.set(code, next); continue; }
+    if (previous.prefecture !== prefecture || previous.municipality !== municipality) throw new Error("Conflicting N03 municipality identity");
+    const polygons = (a: AdministrativeArea): number[][][][] =>
+      a.geometry.type === "Polygon" ? [a.geometry.coordinates as number[][][]] : a.geometry.coordinates as number[][][][];
+    previous.geometry = { type: "MultiPolygon", coordinates: [...polygons(previous), ...polygons(next)] };
+  }
+  return [...merged.values()];
+}

+import { PNG } from "pngjs";
+import { classifyRainPixel } from "../providers/jma/classifyPixel";
+import type { RainIntensityClass } from "../types";
+
+/** Isolated read-only proof: no changes to production >=30mm/h alerts. */
+const AT_LEAST_1: ReadonlySet<RainIntensityClass> = new Set([
+  "1_TO_5", "5_TO_10", "10_TO_20", "20_TO_30",
+  "30_TO_50", "50_TO_80", "GTE_80",
+]);
+
+export type LightRainScanResult = {
+  rainPixels: number;
+  belowOnePixels: number;
+  unknownPixels: number;
+  sampledPixels: number;
+};
+
+/** Full-pixel scan: never use stride=2; unknown pixels are not dry. */
+export function scanJmaTileAtLeast1mm(buffer: Buffer): LightRainScanResult {
+  const png = PNG.sync.read(buffer);
+  const result: LightRainScanResult = {
+    rainPixels: 0, belowOnePixels: 0, unknownPixels: 0, sampledPixels: 0,
+  };
+  for (let i = 0; i < png.data.length; i += 4) {
+    result.sampledPixels++;
+    const classified = classifyRainPixel({
+      r: png.data[i], g: png.data[i + 1], b: png.data[i + 2], a: png.data[i + 3],
+    });
+    if (classified.intensityClass && AT_LEAST_1.has(classified.intensityClass)) result.rainPixels++;
+    else if (classified.intensityClass === "LT_1") result.belowOnePixels++;
+    else if (classified.status === "UNKNOWN_PIXEL") result.unknownPixels++;
+  }
+  return result;
+}
+
+/** Exact rainy pixel coordinates for downstream z10/N03 footprint matching.
+ * Keep every pixel; no stride or center-point-only municipality guess.
+ */
+export function locateJmaRainPixelsAtLeast1mm(buffer: Buffer, zoom: number, tileX: number, tileY: number): Array<{ pixelX: number; pixelY: number; longitude: number; latitude: number }> {
+  const png = PNG.sync.read(buffer);
+  if (png.width !== 256 || png.height !== 256) throw new Error("Unexpected JMA tile dimensions");
+  const points = [];
+  const size = 256 * 2 ** zoom;
+  for (let py = 0; py < png.height; py++) for (let px = 0; px < png.width; px++) {
+    const i = (py * png.width + px) * 4;
+    const result = classifyRainPixel({ r: png.data[i], g: png.data[i + 1], b: png.data[i + 2], a: png.data[i + 3] });
+    if (result.status === "UNKNOWN_PIXEL") throw new Error("Unrecognized JMA pixel");
+    if (!result.intensityClass || !AT_LEAST_1.has(result.intensityClass)) continue;
+    const worldX = tileX * 256 + px + 0.5;
+    const worldY = tileY * 256 + py + 0.5;
+    const n = Math.PI - 2 * Math.PI * worldY / size;
+    points.push({ pixelX: px, pixelY: py, longitude: worldX / size * 360 - 180,
+      latitude: Math.atan(Math.sinh(n)) * 180 / Math.PI });
+  }
+  return points;
+}
