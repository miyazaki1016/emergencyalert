```text
+/** Reused from nationwide rain proof PR #59; isolated geometry matcher. */
+export type HeavyRainPolygon = { coordinates: number[][][] };
+
+export interface AdministrativeArea {
+  code: string;
+  prefecture: string;
+  municipality: string;
+  geometry: {
+    type: "Polygon" | "MultiPolygon";
+    coordinates: number[][][] | number[][][][];
+  };
+}
+
+type Position = [number, number];
+type Bounds = [west: number, south: number, east: number, north: number];
+
+function polygonBounds(polygon: number[][][]): Bounds {
+  let west = Infinity, south = Infinity, east = -Infinity, north = -Infinity;
+  for (const ring of polygon) for (const [x, y] of ring) {
+    west = Math.min(west, x); south = Math.min(south, y);
+    east = Math.max(east, x); north = Math.max(north, y);
+  }
+  return [west, south, east, north];
+}
+
+function boundsIntersect(a: Bounds, b: Bounds) {
+  return a[0] <= b[2] && a[2] >= b[0] && a[1] <= b[3] && a[3] >= b[1];
+}
+
+function pointInRing([x, y]: Position, ring: number[][]) {
+  let inside = false;
+  for (let i = 0, j = ring.length - 1; i < ring.length; j = i++) {
+    const [xi, yi] = ring[i];
+    const [xj, yj] = ring[j];
+    const crosses = yi > y !== yj > y && x < ((xj - xi) * (y - yi)) / (yj - yi) + xi;
+    if (crosses) inside = !inside;
+  }
+  return inside;
+}
+
+function pointInPolygon(point: Position, polygon: number[][][]) {
+  if (!polygon[0] || !pointInRing(point, polygon[0])) return false;
+  return !polygon.slice(1).some((hole) => pointInRing(point, hole));
+}
+
+const EPSILON = 1e-12;
+
+function cross(a: Position, b: Position, c: Position) {
+  return (b[0] - a[0]) * (c[1] - a[1]) - (b[1] - a[1]) * (c[0] - a[0]);
+}
+
+function onSegment(a: Position, b: Position, p: Position) {
+  if (Math.abs(cross(a, b, p)) > EPSILON) return false;
+  return p[0] >= Math.min(a[0], b[0]) - EPSILON &&
+    p[0] <= Math.max(a[0], b[0]) + EPSILON &&
+    p[1] >= Math.min(a[1], b[1]) - EPSILON &&
+    p[1] <= Math.max(a[1], b[1]) + EPSILON;
+}
+
+function segmentsIntersect(a: Position, b: Position, c: Position, d: Position) {
+  const abC = cross(a, b, c);
+  const abD = cross(a, b, d);
+  const cdA = cross(c, d, a);
+  const cdB = cross(c, d, b);
+
+  if (Math.abs(abC) <= EPSILON && onSegment(a, b, c)) return true;
+  if (Math.abs(abD) <= EPSILON && onSegment(a, b, d)) return true;
+  if (Math.abs(cdA) <= EPSILON && onSegment(c, d, a)) return true;
+  if (Math.abs(cdB) <= EPSILON && onSegment(c, d, b)) return true;
+
+  return (abC > EPSILON) !== (abD > EPSILON) &&
+    (cdA > EPSILON) !== (cdB > EPSILON);
+}
+
+function ringsIntersect(a: number[][], b: number[][]) {
+  for (let i = 1; i < a.length; i++) {
+    for (let j = 1; j < b.length; j++) {
+      if (segmentsIntersect(a[i - 1] as Position, a[i] as Position, b[j - 1] as Position, b[j] as Position)) return true;
+    }
+  }
+  return false;
+}
+
+function polygonsIntersect(a: number[][][], b: number[][][]) {
+  const ar = a[0], br = b[0];
+  if (!ar || !br) return false;
+  return ringsIntersect(ar, br) ||
+    pointInPolygon(ar[0] as Position, b) ||
+    pointInPolygon(br[0] as Position, a);
+}
+
+export function rainPolygonIntersectsAdministrativeArea(rain: HeavyRainPolygon, area: AdministrativeArea) {
+  const rainPolygon = rain.coordinates;
+  const areaPolygons = area.geometry.type === "Polygon"
+    ? [area.geometry.coordinates as number[][][]]
+    : area.geometry.coordinates as number[][][][];
+  const rainBounds = polygonBounds(rainPolygon);
+  return areaPolygons.some((polygon) =>
+    boundsIntersect(rainBounds, polygonBounds(polygon)) && polygonsIntersect(rainPolygon, polygon),
+  );
+}
+
+export function affectedAdministrativeAreas(rainPolygons: HeavyRainPolygon[], areas: AdministrativeArea[]) {
+  const rainWithBounds = rainPolygons.map((rain) => ({
+    rain,
+    bounds: polygonBounds(rain.coordinates),
+  }));
+  const areasWithBounds = areas.map((area) => {
+    const polygons = area.geometry.type === "Polygon"
+      ? [area.geometry.coordinates as number[][][]]
+      : area.geometry.coordinates as number[][][][];
+    const prepared = polygons.map((polygon) => ({
+      polygon,
+      bounds: polygonBounds(polygon),
+    }));
+    return {
+      area,
+      polygons: prepared,
+      west: Math.min(...prepared.map(({ bounds }) => bounds[0])),
+      south: Math.min(...prepared.map(({ bounds }) => bounds[1])),
+      east: Math.max(...prepared.map(({ bounds }) => bounds[2])),
+      north: Math.max(...prepared.map(({ bounds }) => bounds[3])),
+    };
+  });
+
+  return areasWithBounds
+    .filter(({ polygons, west, south, east, north }) =>
+      rainWithBounds.some(({ rain, bounds: rainBounds }) => {
+        if (west > rainBounds[2] || east < rainBounds[0] ||
+            south > rainBounds[3] || north < rainBounds[1]) return false;
+        return polygons.some(({ polygon, bounds: areaBounds }) =>
+          boundsIntersect(rainBounds, areaBounds) &&
+          polygonsIntersect(rain.coordinates, polygon),
+        );
+      }),
+    )
+    .map(({ area }) => area);
+}
```

```text
+/** One valid AT Protocol TID per forecast cycle for this publisher. */
+export function lightRainRecordKey(baseTimeMs: number): string {
+  if (!Number.isSafeInteger(baseTimeMs) || baseTimeMs <= 0 || baseTimeMs % 300_000 !== 0) {
+    throw new Error("Invalid five-minute JMA basetime");
+  }
+  let value = (BigInt(baseTimeMs) * 1000n << 10n) | 219n;
+  let key = "";
+  for (let i = 0; i < 13; i++) {
+    key = ALPHABET[Number(value & 31n)] + key;
+    value >>= 5n;
+  }
+  if (value) throw new Error("Basetime exceeds the record-key range");
+  return key;
+}
+

+  now: () => number = Date.now,
+  wait: (ms: number) => Promise<unknown> = delay,
+): Promise<string> {
+  if (!account.appPassword || !/^[a-z0-9.-]+$/.test(account.handle) ||
+    !/^did:(plc|web):[a-z0-9.:%-]+$/.test(account.did)) {
+    throw new Error("Bluesky account and existing app password are required");
+  }
+  if (!post.text || Buffer.byteLength(post.text) > 3000 ||
+    [...new Intl.Segmenter("ja").segment(post.text)].length > 300 ||
+    !/^[234567a-z]{13}$/.test(post.key)) {
+    throw new Error("Invalid Bluesky post");
+  }
+  const assertFresh = () => {
+    if (!Number.isFinite(post.validUntil) || now() >= post.validUntil) {
+      throw new Error("Forecast expired before Bluesky delivery");
+    }
+  };
+  const rpc = async (method: string, input: RequestInit) => {
+    let response: Response;
+    try {
+      response = await fetcher("https://bsky.social/xrpc/" + method, {
+        ...input,
+        redirect: "error",
+        signal: AbortSignal.timeout(10_000),
+      });
+    } catch {
+      throw new RpcError(0, "TransportError");
+    }
+    const body = await response.json().catch(() => null);
+    if (!response.ok || !body) throw new RpcError(response.status, body?.error ?? "");
+    return body;
+  };
+
+  assertFresh();
+  const session = await rpc("com.atproto.server.createSession", {
+    method: "POST",
+    headers: { "content-type": "application/json" },
+    body: JSON.stringify({ identifier: account.handle, password: account.appPassword }),
+  });
+  if (session.did !== account.did || session.handle !== account.handle || typeof session.accessJwt !== "string") {
+    throw new Error("Bluesky authenticated account differs from the expected account");
+  }
+  const headers = { authorization: `Bearer ${session.accessJwt}` };
+  const identity = { repo: account.did, collection: COLLECTION, rkey: post.key };
+  const uri = `at://${account.did}/${COLLECTION}/${post.key}`;
+  const publicUrl = `https://bsky.app/profile/${account.did}/post/${post.key}`;
+  const readExisting = async () => {
+    try {
+      const found = await rpc("com.atproto.repo.getRecord?" + new URLSearchParams(identity), { headers });
+      if (found.uri !== uri || !found.cid || found.value?.text !== post.text) {
+        throw new Error("Bluesky record key is already associated with different content");
+      }
+      return true;
+    } catch (error) {
+      if (error instanceof RpcError && [400, 404].includes(error.status) && error.code === "RecordNotFound") return false;
+      throw error;
+    }
+  };
+  const record = { $type: COLLECTION, text: post.text, createdAt: new Date(now()).toISOString(), langs: ["ja"] };
+  for (let attempt = 0; attempt < 3; attempt++) {
+    if (await readExisting()) return publicUrl;
+    assertFresh();
+    try {
+      const created = await rpc("com.atproto.repo.createRecord", {
+        method: "POST",
+        headers: { ...headers, "content-type": "application/json" },
+        body: JSON.stringify({ ...identity, record, validate: true }),
+      });
+      if (created.uri === uri && created.cid) return publicUrl;
+      throw new RpcError(0, "UnconfirmedWrite");
+    } catch (error) {
+      // Reconcile ambiguous responses and races before considering another write.
+      if (await readExisting()) return publicUrl;
+      if (!(error instanceof RpcError) || ![0, 429, 500, 502, 503, 504].includes(error.status) || attempt === 2) throw error;
+      await wait(1000 * (attempt + 1));
+    }
+  }
+  throw new Error("Bluesky delivery unconfirmed");
+}
```

```text
+  if (!/^\d{2}$/.test(expectedPrefectureCode)) throw new Error("Invalid prefecture code");

+  });
```

```text
+  prefectures: number;

+  const codes = new Set<string>();
+  const prefectureNames = new Set<string>();
+  for (const prefectureCode of ALL_PREFECTURE_CODES) {
+    const file = join(root, prefectureCode + ".areas.json");

+      missing.push(prefectureCode);
+      continue;
+    }
+    const loaded = loadPreparedN03ForProof(file, prefectureCode);

+      if (codes.has(area.code)) throw new Error("Duplicate nationwide N03 code " + area.code);
+      codes.add(area.code);
+      areas.push(area);
+    }
+  }
+  return { complete: missing.length === 0, prefectures: 47 - missing.length, municipalities: codes.size, missing, areas };

+  const coverage = loadNationwidePreparedN03(root);
```

```text
+type N03Feature = { type: "Feature"; properties?: Record<string, unknown> | null; geometry: N03Geometry | null };
+export type N03FeatureCollection = { type: "FeatureCollection"; features: N03Feature[] };

+/** Reused PR #59 official N03 feature conversion, isolated from production. */
+export function parseOfficialN03ForLightRain(collection: N03FeatureCollection): AdministrativeArea[] {
+  if (collection.type !== "FeatureCollection" || !Array.isArray(collection.features)) throw new Error("Invalid N03 GeoJSON");
+  const merged = new Map<string, AdministrativeArea>();
+  for (const feature of collection.features) {
+    if (!feature.geometry || !["Polygon", "MultiPolygon"].includes(feature.geometry.type)) continue;
+    const p = feature.properties ?? {};
+    const prefecture = typeof p.N03_001 === "string" ? p.N03_001.trim() : "";
+    const city = typeof p.N03_004 === "string" ? p.N03_004.trim() : "";
+    const ward = typeof p.N03_005 === "string" ? p.N03_005.trim() : "";
+    const code = typeof p.N03_007 === "string" ? p.N03_007.trim() : "";
+    if (!prefecture || !city || !/^\d{5}$/.test(code)) continue;
+    const municipality = city + ward;
+    const next: AdministrativeArea = { code, prefecture, municipality, geometry: feature.geometry };
+    const previous = merged.get(code);
+    if (!previous) { merged.set(code, next); continue; }
+    if (previous.prefecture !== prefecture || previous.municipality !== municipality) throw new Error("Conflicting N03 municipality identity");
+    const polygons = (a: AdministrativeArea): number[][][][] =>
+      a.geometry.type === "Polygon" ? [a.geometry.coordinates as number[][][]] : a.geometry.coordinates as number[][][][];
+    previous.geometry = { type: "MultiPolygon", coordinates: [...polygons(previous), ...polygons(next)] };
+  }
```

```text
+  rainPixels: number;
+  belowOnePixels: number;
+  unknownPixels: number;
+  sampledPixels: number;
+};

+/** Full-pixel scan: never use stride=2; unknown pixels are not dry. */

+  const png = PNG.sync.read(buffer);
+  const result: LightRainScanResult = {
+    rainPixels: 0, belowOnePixels: 0, unknownPixels: 0, sampledPixels: 0,
+  };
+  for (let i = 0; i < png.data.length; i += 4) {
+    result.sampledPixels++;
+    const classified = classifyRainPixel({
+      r: png.data[i], g: png.data[i + 1], b: png.data[i + 2], a: png.data[i + 3],
+    });
+    if (classified.intensityClass && AT_LEAST_1.has(classified.intensityClass)) result.rainPixels++;
+    else if (classified.intensityClass === "LT_1") result.belowOnePixels++;
+    else if (classified.status === "UNKNOWN_PIXEL") result.unknownPixels++;
+  }
+  return result;
+}
+
+/** Exact rainy pixel coordinates for downstream z10/N03 footprint matching.
+ * Keep every pixel; no stride or center-point-only municipality guess.
+ */
+export function locateJmaRainPixelsAtLeast1mm(buffer: Buffer, zoom: number, tileX: number, tileY: number): Array<{ pixelX: number; pixelY: number; longitude: number; latitude: number }> {
+  const png = PNG.sync.read(buffer);
+  if (png.width !== 256 || png.height !== 256) throw new Error("Unexpected JMA tile dimensions");
+  const points = [];
+  const size = 256 * 2 ** zoom;
+  for (let py = 0; py < png.height; py++) for (let px = 0; px < png.width; px++) {
+    const i = (py * png.width + px) * 4;
+    const result = classifyRainPixel({ r: png.data[i], g: png.data[i + 1], b: png.data[i + 2], a: png.data[i + 3] });
+    if (result.status === "UNKNOWN_PIXEL") throw new Error("Unrecognized JMA pixel");
+    if (!result.intensityClass || !AT_LEAST_1.has(result.intensityClass)) continue;
+    const worldX = tileX * 256 + px + 0.5;
+    const worldY = tileY * 256 + py + 0.5;
+    const n = Math.PI - 2 * Math.PI * worldY / size;
+    points.push({ pixelX: px, pixelY: py, longitude: worldX / size * 360 - 180,
+      latitude: Math.atan(Math.sinh(n)) * 180 / Math.PI });
+  }
+  return points;
+}
```

```text
+export type ForecastPostCandidate<Strength extends number = 30 | 50 | 80> = {
+  key: string;
+  text: string;
+  municipalityCode: string;
+  validTime: string;
+  intensity: Strength;
+};
+type ForecastArea<Strength extends number> = Omit<
+  NationalForecastSnapshot["frames"][number]["municipalities"][number], "minimumMmPerHour"
+> & { minimumMmPerHour: Strength };
+type ForecastInput<Strength extends number> = Omit<NationalForecastSnapshot, "frames"> & {
+  frames: { validTime: string; municipalities: ForecastArea<Strength>[] }[];
+};
+
+/** Shared validated snapshot is the only source; never fetch JMA separately. */
+export function buildForecastPostCandidates<Strength extends 1 | 30 | 50 | 80>(
+  snapshot: ForecastInput<Strength>,
+): ForecastPostCandidate<Strength>[] {
+  const strongest = new Map<string, { area: ForecastArea<Strength>; validTime: string }>();
+  for (const frame of snapshot.frames) for (const area of frame.municipalities) {
+    const old = strongest.get(area.code);
+    if (!old || area.minimumMmPerHour > old.area.minimumMmPerHour ||
+      (area.minimumMmPerHour === old.area.minimumMmPerHour && forecastTime(frame.validTime) < forecastTime(old.validTime))) {
+      strongest.set(area.code, { area, validTime: frame.validTime });
+    }
+  }
+  return [...strongest.values()].sort((a, b) =>
+    b.area.minimumMmPerHour - a.area.minimumMmPerHour ||
+    forecastTime(a.validTime) - forecastTime(b.validTime) ||
+    a.area.code.localeCompare(b.area.code)
+  ).map(({ area, validTime }) => {
+    const title = area.minimumMmPerHour === 1 ? "全国降雨予報" : "全国大雨予報";
+    const tags = area.minimumMmPerHour === 1 ? "#アメふる" : "#大雨 #アメふる";
+    const text = `【アメふる ${title}】\n${area.prefecture} ${area.municipality}\n${new Intl.DateTimeFormat("ja-JP", {timeZone:"Asia/Tokyo",month:"numeric",day:"numeric",hour:"2-digit",minute:"2-digit",hourCycle:"h23"}).format(forecastTime(validTime))}ごろ、1時間に${area.minimumMmPerHour}mm以上の雨の予報\n${tags}`;
+    const key = createHash("sha256").update(JSON.stringify([snapshot.n03Version, area.code, validTime, area.minimumMmPerHour])).digest("hex");
+    return { key, text, municipalityCode: area.code, validTime, intensity: area.minimumMmPerHour };
+  });

+  if (previous) for (const frame of previous.frames) for (const area of frame.municipalities)
+    oldMax.set(area.code, Math.max(oldMax.get(area.code) ?? 0, area.minimumMmPerHour));

+    previous !== null && row.intensity > (oldMax.get(row.municipalityCode) ?? 0));

+};

+  previous: NationalForecastSnapshot | null,
+): ForecastPublishPlan | null {
+  if (previous && forecastTime(current.basetime) <= forecastTime(previous.basetime)) return null;

+  };

+  /** Release only an uncompleted claim; retry policy belongs to the durable adapter. */
+  release(basetime: string): Promise<void>;
+}

+ * The caller must durably record individual delivery outcomes before calling complete().

+export async function prepareClaimedForecastPublication(

+  store: ForecastPublicationStore,
+): Promise<{ plan: ForecastPublishPlan; complete: () => Promise<void>; release: () => Promise<void> } | null> {
+  if (!(await store.claim(current.basetime))) return null;

+      return null;
+    }

+      plan,
+      complete: () => store.complete(current.basetime, current),
+      release: () => store.release(current.basetime),
+    };
+  } catch (error) {
+    await store.release(current.basetime);
+    throw error;
+  }
```

```text
+    for (let t = from + FIVE_MINUTES; t < to; t += FIVE_MINUTES) {
+      const missing = new Date(t).toISOString();
+      missingBasetimes.push(missing);
+      console.error(JSON.stringify({ event: "MISSING_JMA_BASETIME", basetime: missing, nextBasetime: base, recovered: false, publication: false }));
+    }
+  };
+  const update = async () => {
+    const scanStarted = performance.now();
+    // Stop this slot before the next JMA publication window; never recover old slots.
+    const slotEnd = Math.floor(Date.now() / FIVE_MINUTES) * FIVE_MINUTES + FIVE_MINUTES + 90_000;
+    const controller = new AbortController();
+    const timer = setTimeout(() => controller.abort(new Error("JMA_SLOT_DEADLINE_EXCEEDED")), Math.max(0, slotEnd - Date.now()));
+    try {
+    const targets = await fetchForecastTargetTimes((input, init) =>
+      fetch(input, { ...init, signal: AbortSignal.any([controller.signal, AbortSignal.timeout(15_000)]) }));
+    const frames = selectFrames(targets, Date.now());
+    if (frames[0].basetime === previousBase) return;
+    controller.signal.throwIfAborted();
+    const results = [];
+    // Two frames share the cycle budget. Await both before stopping so the next
+    // update does not overlap a failed cycle's remaining requests.
+    for (let i = 0; i < frames.length; i += 2) {
+      const batch = await Promise.allSettled(
+        frames.slice(i, i + 2).map(frame => refineFrame(frame, indexed, areaByCode, controller.signal)),
+      );
+      for (const result of batch) {
+        if (result.status === "rejected") throw result.reason;
+        results.push(result.value);
+      }
+    }
+
+    if (Date.now() - forecastTime(frames[0].basetime) >= 2 * FIVE_MINUTES) {
+      throw new Error("JMA forecast expired before the scan finished");
+    }
+    console.log(JSON.stringify({
+      mode: "READ_ONLY_Z10_REAL_JMA_N03",
+      basetime: frames[0].basetime,
+      checkedAt: new Date().toISOString(),
+      baseAgeMs: Date.now() - forecastTime(frames[0].basetime),
+      prefectures: coverage.prefectures,
+      municipalityDatasetSize: coverage.municipalities,
+      thresholdMmPerHour: 1,
+      timingMs: { n03LoadAndIndex, total: Math.round(n03LoadAndIndex + performance.now() - scanStarted) },
+      frames: results,
+      publication: false,
+      productionWrites: false,
+      blueskyPosts: false,
+    }));
+    // Record a gap only after the new cycle has actually completed. A failed
+    // cycle must not be mistaken for successful progress.
+    recordGap(frames[0].basetime);
+    previousBase = frames[0].basetime;
+    successfulUpdates++;
+    } finally { clearTimeout(timer); }
+  };
+
+  if (!process.argv.includes("--watch")) return update();
+  const watchCount = Number(process.env.JMA_WATCH_SUCCESS_COUNT || "0");
+  if (!Number.isSafeInteger(watchCount) || watchCount < 0) throw new Error("Invalid watch success count");
+  while (watchCount === 0 || successfulUpdates < watchCount) {
+    try {
+      await update();
+    } catch (error) {
+      console.error(JSON.stringify({ event: "JMA_SLOT_FAILED", at: new Date().toISOString(), error: String(error), recovery: false, publication: false }));
+    }
+    if (watchCount > 0 && successfulUpdates >= watchCount) {
+      if (missingBasetimes.length) {
+        throw new Error("INCOMPLETE_FIVE_MINUTE_COVERAGE: " + missingBasetimes.join(", "));
+      }
+      return;
+    }
+    // Let JMA publish the new cycle before polling the next five-minute slot.
+    const now = Date.now();
+    const offset = 120_000;
+    await delay(FIVE_MINUTES - (now - offset) % FIVE_MINUTES);
+  }
+}
+
+main().catch(error => { console.error(error); process.exitCode = 1; });

```
