+-- Add daily completion summaries to the existing private Resend monitor.
+-- The production dispatcher, 18:00 job, delivery identity and rain jobs are untouched.
+create function jaspic_monitor.scan_completions(p_now timestamptz default now()) returns void
+language plpgsql set search_path='' as $$
+declare c jaspic_monitor.config; r record; day_key text; h integer; s integer;
+begin
+ select * into strict c from jaspic_monitor.config where id;
+ if not c.enabled then return; end if;
+ for r in
+  select distinct on ((started_at at time zone 'Asia/Tokyo')::date) *
+  from public.jaspic_runs runs
+  where status='SUCCEEDED' and finished_at is not null and finished_at<=p_now
+   and started_at>=c.activated_at and started_at>=p_now-interval '48 hours'
+   and finished_at>=started_at
+   -- Same scheduled-start window used by the existing missing-run watchdog.
+   and (started_at at time zone 'Asia/Tokyo')::time>=time '18:00'
+   and (started_at at time zone 'Asia/Tokyo')::time<time '18:10'
+   and exists(select 1 from cron.job_run_details d join cron.job j using(jobid)
+    where j.jobname='jaspic-production-1800-jst' and d.status='succeeded'
+     and d.start_time>=(((runs.started_at at time zone 'Asia/Tokyo')::date+time '18:00') at time zone 'Asia/Tokyo')
+     and d.start_time<=runs.started_at)
+  order by (started_at at time zone 'Asia/Tokyo')::date,started_at,id
+ loop
+  day_key=to_char(r.started_at at time zone 'Asia/Tokyo','YYYY-MM-DD');
+  -- A missing/malformed count is not evidence that zero messages were sent.
+  if r.result->'sent'->>'hiroshima' is null or r.result->'sent'->>'shiga' is null
+   or r.result->'sent'->>'hiroshima' !~ '^(0|[1-9][0-9]?)$'
+   or r.result->'sent'->>'shiga' !~ '^(0|[1-9][0-9]?)$' then
+   perform jaspic_monitor.enqueue('completion-invalid:'||day_key,'完了結果確認','jaspic/dispatch',
+    'COMPLETION_RESULT_INVALID',r.finished_at,'成功記録の県別送信件数を読み取れません。実行ID: '||r.id);
+   continue;
+  end if;
+  h=(r.result->'sent'->>'hiroshima')::integer;
+  s=(r.result->'sent'->>'shiga')::integer;
+  if h+s>20 then
+   perform jaspic_monitor.enqueue('completion-invalid:'||day_key,'完了結果確認','jaspic/dispatch',
+    'COMPLETION_RESULT_INVALID',r.finished_at,'成功記録の送信件数が配信上限を超えています。実行ID: '||r.id);
+   continue;
+  end if;
+  perform jaspic_monitor.enqueue('completion:'||day_key,'配信完了','jaspic/dispatch','DISPATCH_SUCCEEDED',r.finished_at,
+   '18時の配信処理が正常終了しました。'||
+   E'\n広島県: '||h||E' 件\n滋賀県: '||s||E' 件\n合計: '||(h+s)||' 件'||
+   case when h+s=0 then E'\n今回は送信対象の新規・更新情報が0件でした。両県の確認処理は正常に完了しています。' else '' end||
+   E'\n処理開始（日本時間）: '||to_char(r.started_at at time zone 'Asia/Tokyo','YYYY-MM-DD HH24:MI:SS')||
+   E'\n処理完了（日本時間）: '||to_char(r.finished_at at time zone 'Asia/Tokyo','YYYY-MM-DD HH24:MI:SS')||
+   E'\n実行ID: '||r.id);
+ end loop;
+end $$;
+
+create function jaspic_monitor.completion_tick() returns void
+language plpgsql set search_path='' as $$
+begin
+ -- Keep completion-summary errors from stopping the existing error/expiry sender.
+ begin
+  perform jaspic_monitor.scan_completions();
+ exception when others then
+  perform jaspic_monitor.enqueue('completion-scan-failed:'||to_char(now() at time zone 'Asia/Tokyo','YYYY-MM-DD'),
+   '完了通知監視エラー','jaspic-monitor','COMPLETION_SCAN_FAILED',now(),
+   '完了メールの生成処理で異常が発生しました。既存の認証期限・エラー監視は継続します。');
+ end;
+ perform jaspic_monitor.tick();
+end $$;
+revoke all on function jaspic_monitor.scan_completions(timestamptz),jaspic_monitor.completion_tick()
+ from public,anon,authenticated,service_role;
+-- Replace only the monitor's command, preserving its name and five-minute schedule.
+select cron.schedule('jaspic-admin-monitor-every-5-minutes','*/5 * * * *','select jaspic_monitor.completion_tick();');

#!/usr/bin/env python3
"""Read-only GitHub development inventory. No branch/PR/Issue mutations."""
import datetime as dt
import json
import os
import sys
import urllib.error
import urllib.parse
import urllib.request

REPO = os.environ["GITHUB_REPOSITORY"]
TOKEN = os.environ["GITHUB_TOKEN"]
NOW = dt.datetime.now(dt.timezone.utc)
STALE_DAYS = int(os.environ.get("STALE_DAYS", "14"))
API = "https://api.github.com"

def get(path):
    req = urllib.request.Request(API + path, headers={
        "Authorization": "Bearer " + TOKEN,
        "Accept": "application/vnd.github+json",
        "X-GitHub-Api-Version": "2022-11-28",
        "User-Agent": "development-inventory-readonly",
    })
    with urllib.request.urlopen(req, timeout=30) as response:
        return json.load(response)

def pages(path):
    result = []
    page = 1
    while True:
        separator = "&" if "?" in path else "?"
        batch = get(path + separator + "per_page=100&page=" + str(page))
        if not isinstance(batch, list):
            raise RuntimeError("Unexpected API response")
        result.extend(batch)
        if len(batch) < 100:
            return result
        page += 1

def age(timestamp):
    if not timestamp:
        return None
    updated = dt.datetime.fromisoformat(timestamp.replace("Z", "+00:00"))
    return (NOW - updated).days

def main():
    repo = urllib.parse.quote(REPO, safe="/")
    metadata = get("/repos/" + repo)
    default = metadata["default_branch"]
    branches = pages("/repos/" + repo + "/branches")
    all_prs = pages("/repos/" + repo + "/pulls?state=all")
    prs = [p for p in all_prs if p["state"] == "open"]
    issues = [item for item in pages("/repos/" + repo + "/issues?state=open") if "pull_request" not in item]
    pr_heads = {p["head"]["ref"] for p in prs if p["head"]["repo"] and p["head"]["repo"]["full_name"] == REPO}
    historical_heads = {p["head"]["ref"] for p in all_prs if p["head"]["repo"] and p["head"]["repo"]["full_name"] == REPO}
    findings = []
    for branch in branches:
        name = branch["name"]
        if name == default:
            continue
        # No-PR does NOT mean unfinished. Report only as a review candidate.
        if name not in pr_heads:
            findings.append({"type": "branch_without_open_pr", "name": name,
                             "historical_pr_found": name in historical_heads,
                             "url": "https://github.com/" + REPO + "/tree/" + urllib.parse.quote(name, safe="")})
    stale_prs = [{"number": p["number"], "title": p["title"], "days": age(p["updated_at"]),
                  "url": p["html_url"]} for p in prs if age(p["updated_at"]) is not None and age(p["updated_at"]) >= STALE_DAYS]
    stale_issues = [{"number": i["number"], "title": i["title"], "days": age(i["updated_at"]),
                     "url": i["html_url"]} for i in issues if age(i["updated_at"]) is not None and age(i["updated_at"]) >= STALE_DAYS]
    report = {"repository": REPO, "generated_at": NOW.isoformat(), "default_branch": default,
              "branch_count": len(branches), "open_pr_count": len(prs),
              "open_issue_count": len(issues), "historical_pr_count": len(all_prs), "review_candidates": findings,
              "stale_prs": stale_prs, "stale_issues": stale_issues,
              "caveat": "Candidates are NOT proof of unfinished work. Compare main, merged/closed PRs, and canonical ledger before action."}
    # Stable snapshot for comparison across successful scheduled/manual runs.
    report["snapshot"] = {
        "branches": {b["name"]: b.get("commit", {}).get("sha") for b in branches},
        "prs": {str(p["number"]): {"state": ("merged" if p.get("merged_at") else p["state"]),
                                  "head": p["head"].get("sha"), "updated_at": p["updated_at"],
                                  "title": p.get("title", ""), "body": (p.get("body") or "")[:2000],
                                  "head_ref": p["head"].get("ref", "")}
                for p in all_prs},
        "issues": {str(i["number"]): {"state": "open", "updated_at": i["updated_at"]} for i in issues},
    }
    os.makedirs("inventory-output", exist_ok=True)
    with open("inventory-output/report.json", "w", encoding="utf-8") as f:
        json.dump(report, f, ensure_ascii=False, indent=2)
    summary = ["# Development inventory (read-only)", "",
               "Repository: " + REPO, "Generated (UTC): " + NOW.isoformat(), "",
               "| Metric | Count |", "|---|---:|",
               "| Branches including default | " + str(len(branches)) + " |",
               "| Open PRs | " + str(len(prs)) + " |",
               "| Open issues | " + str(len(issues)) + " |",
               "| Branches without open PR (review only) | " + str(len(findings)) + " |",
               "| Stale PRs (>= " + str(STALE_DAYS) + " days) | " + str(len(stale_prs)) + " |",
               "| Stale issues | " + str(len(stale_issues)) + " |", "",
               "**Do not mark branches unfinished or delete them based on this report.**",
               "Consult the project's canonical ledger, closed/merged PRs and main code first.", "",
               "## Stale PRs"]
    summary.extend("- [#" + str(p["number"]) + "](" + p["url"] + ") (" + str(p["days"]) + "d): " + p["title"].replace("\n", " ") for p in stale_prs)
    summary.append("\n## Stale issues")
    summary.extend("- [#" + str(i["number"]) + "](" + i["url"] + ") (" + str(i["days"]) + "d): " + i["title"].replace("\n", " ") for i in stale_issues)
    summary.append("\nBranch candidates are available in report.json artifact.")
    text = "\n".join(summary) + "\n"
    with open("inventory-output/summary.md", "w", encoding="utf-8") as f:
        f.write(text)
    with open(os.environ.get("GITHUB_STEP_SUMMARY", "/dev/null"), "a", encoding="utf-8") as f:
        f.write(text)
    print(text)

if __name__ == "__main__":
    try:
        main()
    except (urllib.error.URLError, KeyError, ValueError, RuntimeError) as error:
        print("Inventory failed: " + str(error), file=sys.stderr)
        sys.exit(1)

#!/usr/bin/env python3
"""Generate read-only inventory change candidates from the prior successful artifact."""
import io
import json
import os
import urllib.parse
import urllib.error
import urllib.request
import urllib.error
import zipfile
from pathlib import Path

API = "https://api.github.com"
REPO = os.environ.get("GITHUB_REPOSITORY", "")
TOKEN = os.environ.get("GITHUB_TOKEN", "")
CURRENT_RUN = str(os.environ.get("GITHUB_RUN_ID", ""))

class SafeRedirect(urllib.request.HTTPRedirectHandler):
    """Do not forward GitHub bearer tokens to external artifact storage."""
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        redirected = super().redirect_request(req, fp, code, msg, headers, newurl)
        if redirected is not None and urllib.parse.urlsplit(newurl).hostname != urllib.parse.urlsplit(req.full_url).hostname:
            redirected.remove_header("Authorization")
            redirected.unredirected_hdrs.pop("Authorization", None)
        return redirected

OPENER = urllib.request.build_opener(SafeRedirect)

def api(path, binary=False):
    req = urllib.request.Request(API + path, headers={
        "Authorization": "Bearer " + TOKEN,
        "Accept": "application/vnd.github+json",
        "X-GitHub-Api-Version": "2022-11-28",
        "User-Agent": "development-inventory-delta-readonly",
    })
    try:
        response = OPENER.open(req, timeout=30)
    except urllib.error.HTTPError as error:
        # Expose the failed endpoint *path*, never signed query parameters or tokens.
        location = urllib.parse.urlsplit(error.url)
        raise RuntimeError("GitHub artifact lookup HTTP " + str(error.code) +
                           " at " + location.hostname + location.path) from error
    with response:
        return response.read() if binary else json.load(response)

def previous_snapshot():
    repo = urllib.parse.quote(REPO, safe="/")
    runs = api("/repos/" + repo + "/actions/workflows/development-inventory.yml/runs?status=success&per_page=30")
    for run in runs.get("workflow_runs", []):
        if str(run["id"]) == CURRENT_RUN or run["event"] not in ("schedule", "workflow_dispatch"):
            continue
        artifacts = api("/repos/" + repo + "/actions/runs/" + str(run["id"]) + "/artifacts?per_page=100")
        for item in artifacts.get("artifacts", []):
            if item["name"] != "development-inventory" or item.get("expired"):
                continue
            archive = api("/repos/" + repo + "/actions/artifacts/" + str(item["id"]) + "/zip", binary=True)
            with zipfile.ZipFile(io.BytesIO(archive)) as z:
                with z.open("report.json") as f:
                    previous = json.load(f)
            if previous.get("repository") == REPO and "snapshot" in previous:
                return previous, run["id"]
    return None, None

def compare(previous, current):
    changes = []
    for group in ("branches", "prs", "issues"):
        before = previous.get("snapshot", {}).get(group, {})
        after = current["snapshot"][group]
        for key in sorted(set(before) | set(after)):
            if key not in before:
                kind = "added"
            elif key not in after:
                kind = "absent_from_snapshot"
            elif before[key] != after[key]:
                kind = "changed"
            else:
                continue
            changes.append({"type": group, "id": key, "change": kind,
                            "before": before.get(key), "after": after.get(key)})
    return changes


def classify_candidates(changes, index_text):
    """Conservative review-only matching; PR files are evidence, never proof."""
    import re
    rows = re.findall(r"^\|\s*([AM]-\d+)\s*\|\s*([^|]+)\|", index_text, re.M)
    suggestions = []
    for change in changes:
        identifier = str(change["id"])
        details = change.get("after") or change.get("before") or {}
        # Branch snapshots contain SHA strings, while PR snapshots contain dicts.
        # Never treat a branch SHA as PR metadata.
        if not isinstance(details, dict):
            details = {}
        metadata = " ".join(str(details.get(k, "")) for k in ("title", "body", "head_ref"))
        filenames = []
        file_status = "not_applicable"
        if change["type"] == "prs" and identifier.isdigit() and REPO:
            try:
                path = "/repos/" + urllib.parse.quote(REPO, safe="/") + "/pulls/" + identifier + "/files?per_page=100"
                files = api(path)
                filenames = [item["filename"] for item in files if "filename" in item]
                file_status = "partial_first_100" if len(files) == 100 else "complete"
            except (RuntimeError, urllib.error.URLError, ValueError, KeyError):
                file_status = "unavailable"
        text = (identifier + " " + metadata + " " + " ".join(filenames)).lower()
        tokens = set(re.findall(r"[a-z0-9]{4,}", text.replace("_", "-")))
        tokens.update(re.findall(r"[\u3040-\u30ff\u4e00-\u9fff]{2,}", text))
        matches = []
        for case_id, title in rows:
            title_tokens = set(re.findall(r"[a-z0-9]{4,}", title.lower()))
            title_tokens.update(re.findall(r"[\u3040-\u30ff\u4e00-\u9fff]{2,}", title))
            if tokens & title_tokens:
                matches.append(case_id)
        # Generic infrastructure PRs are not forced into product work items.
        infrastructure = any(x in text for x in ("development_inventory", "development-inventory", "inventory-triage"))
        if infrastructure:
            matches = []
        suggestions.append({
            "source_type": change["type"], "source_id": identifier,
            "change": change["change"], "candidate_case_ids": matches,
            "classification": ("automation_maintenance" if infrastructure else
                               "possible_existing" if matches else "needs_review"),
            "ledger_action": "review_only", "proposed_status": None,
            "evidence": "PR metadata and changed file names; unverified token hints",
            "file_evidence": {"status": file_status, "paths": filenames[:100]}
        })
    return suggestions

def write_triage(output, result):

    (output / "ledger-review-proposals.md").write_text("\n".join(lines) + "\n", encoding="utf-8")

def main():
    output = Path("inventory-output")
    current = json.loads((output / "report.json").read_text(encoding="utf-8"))
    error_note = None
    try:
        previous, run_id = previous_snapshot()
    except (RuntimeError, urllib.error.HTTPError, urllib.error.URLError, zipfile.BadZipFile, KeyError, ValueError) as error:
        # Never discard today's complete inventory because a prior artifact cannot be read.
        # Explicitly report the comparison failure; do NOT claim zero changes.
        previous, run_id = None, None
        error_note = "Previous report unavailable: " + type(error).__name__
        if isinstance(error, urllib.error.HTTPError):
            error_note += " (HTTP " + str(error.code) + ")"
        if isinstance(error, RuntimeError):
            error_note += " (" + str(error) + ")"
    if error_note:
        result = {"status": "comparison_unavailable", "previous_run_id": None, "changes": [],
                  "note": error_note + "; delta NOT checked. Current inventory saved."}
    elif previous is None:
        result = {"status": "baseline_only", "previous_run_id": None, "changes": [],
                  "note": "No prior compatible successful artifact; do not infer no changes."}
    else:
        result = {"status": "compared", "previous_run_id": run_id,
                  "changes": compare(previous, current),
                  "note": "Candidates only. No automatic ledger status or completion decisions."}
    (output / "delta.json").write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    lines = ["# Development inventory change candidates", "",
             "Status: " + result["status"],
             "Previous run: " + str(run_id or "none"),
             "Candidates: " + str(len(result["changes"])), "",
             result["note"], "",
             "**Read-only suggestions; verify main, PR history and ledger before updating.**", ""]
    for c in result["changes"]:
        lines.append("- " + c["type"] + " `" + c["id"].replace("`", "") + "`: " + c["change"])
    (output / "ledger-update-candidates.md").write_text("\n".join(lines) + "\n", encoding="utf-8")
    write_triage(output, result)
    print("\n".join(lines))

if __name__ == "__main__":
    main()

def normal(text):
    return unicodedata.normalize('NFKC', str(text or '')).lower()


def concepts(text):
    value = normal(text)
    found = {name for name, aliases in CONCEPTS.items() if any(normal(alias) in value for alias in aliases)}
    if re.search(r'テスト.{0,12}(?:認識|検出)', value):

def tokens(text):
    value = normal(text)
    out = set(re.findall(r'[a-z][a-z0-9_:-]{2,}', value)) - STOP
    for run in re.findall(r'[一-龯ぁ-んァ-ヶー0-9]+', value):

def case_text(case):
    curated = case.get('curated') or {}
    primary = ' '.join([case.get('title', ''), curated.get('summary', ''), curated.get('conclusion', '')])
    # Long PRs must not win merely because they contain every project keyword.
    secondary = ' '.join([case.get('body', '')[:12000], ' '.join(case.get('symbols', [])),
                          ' '.join(f['path'] for f in case.get('files', []))])
    return primary, secondary


def evidence_links(case, repo):

    result = [root+'/blob/development-history/docs/work-history/cases/'+case['id']+'.md']
    if case.get('number'):
        result.append(root+'/pull/'+str(case['number']))
    result.extend((case.get('curated') or {}).get('evidence', []))
    result.extend(f.get('url') for f in case.get('files', []) if f.get('url'))
    return list(dict.fromkeys(u for u in result if isinstance(u, str) and u.startswith(root+'/')))


def match_cases(state, task, paths=(), exclude=None, limit=8):
    excluded = {exclude} if isinstance(exclude, str) else set(exclude or [])
    cases = [c for c in state['cases'].values() if c['id'] not in excluded]
    query = tokens(task)
    query_concepts = concepts(task)
    if not query and not paths:
        return []
    documents = [(c, *case_text(c)) for c in cases]
    document_tokens = [(c, p, s, tokens(p), tokens(s)) for c, p, s in documents]
    frequency = Counter(t for _, _, _, p, s in document_tokens for t in (p | s))
    weights = {t: math.log(1 + (len(cases)+1)/(frequency[t]+1)) for t in query}
    total = sum(weights.values()) or 1
    results = []
    for c, primary, secondary, pt, st in document_tokens:
        common = query & (pt | st)
        # Primary findings count in full; body and filenames contribute 30%.
        score = 70*sum(weights[t]*(1 if t in pt else .3) for t in common)/total
        matched_concepts = query_concepts & concepts(primary+' '+secondary)
        primary_concepts = query_concepts & concepts(primary)
        score += min(30, 10*len(primary_concepts))
        overlap = sorted(set(paths) & {f['path'] for f in c.get('files', [])})
        if overlap:
            score += 38
        explicit = bool(c.get('number') and re.search(r'(?:pr\s*#?\s*|#)'+str(c['number'])+r'(?![0-9])', normal(task)))
        if explicit:
            score += 80
        # A generic topic alone (e.g. "notification") is not enough to block work.
        specific = matched_concepts - {'比較検証', '性能', '雨の検知', '通知配信', '承認'}
        required = explicit or bool(overlap) or (score >= 24 and len(primary_concepts) >= 2 and bool(specific))
        if score < 13 and not required:
            continue
        links = evidence_links(c, state['repository'])
        results.append({'id':c['id'], 'title':clean(c['title']), 'score':round(score, 2),
                        'review_required':required, 'matched_concepts':sorted(matched_concepts),
                        'matched_paths':overlap, 'matched_terms':sorted(common, key=lambda t:-weights[t])[:12],
                        'summary':clean((c.get('curated') or {}).get('summary') or c.get('purpose') or c['title']),
                        'conclusion':clean((c.get('curated') or {}).get('conclusion') or '結論未確認。根拠のコード・検証資料を読む'),
                        'remaining':clean(c.get('remaining') or 'PR・検証資料の未解決事項を確認'),
                        'next':clean(c.get('next') or '関連PRと既存実装を照合する'),

    results.sort(key=lambda c:(-c['score'], not c['id'].startswith('pr-'), c['id']))

    seen_heads = set()

def coverage_problems(state, now=None, max_age_hours=48):
    now = now or dt.datetime.now(dt.timezone.utc)
    issues = []
    try:
        generated = dt.datetime.fromisoformat(state['generated_at'].replace('Z', '+00:00'))
        age = (now-generated).total_seconds()/3600
        if age > max_age_hours or age < -1:
            issues.append('履歴の更新日時が古い、または未来。日次棚卸しを更新して再照合する')
    except (KeyError, TypeError, ValueError):
        issues.append('履歴の更新日時を確認できない')
    cov = state.get('coverage') or {}
    if cov.get('errors') or not isinstance(cov.get('errors'), list):
        issues.append('履歴の収集エラーまたは収集状態不明')
    active = [c for c in state.get('cases', {}).values() if not c.get('archived')]
    prs = sum(c['id'].startswith('pr-') for c in active)
    if not prs or prs != cov.get('prs'):
        issues.append('全PRの収録件数が一致しない、またはPRがない')
    if sum(c['id'].startswith('branch-') for c in active) != cov.get('branches'):
        issues.append('全作業ブランチの収録件数が一致しない')
    if any(c.get('error') or c.get('state') == '取得不能' for c in active):
        issues.append('証拠取得に失敗した案件がある')
    return issues

            unresolved.append(case['id'])
    problems = coverage_problems(state, now, max_age_hours)
    status = 'INCOMPLETE' if problems else 'REVIEW_REQUIRED' if unresolved else 'REVIEW_RECORDED' if required else 'NO_STRONG_MATCH'

def git(*args):
    return subprocess.check_output(['git', *args], text=True, stderr=subprocess.DEVNULL, timeout=120)


def load_history(path=None, repo=None, refresh=True):
    if path:
        return json.loads(Path(path).read_text()), None
    if refresh:
        git('fetch', '--quiet', '--no-tags', 'origin', '+refs/heads/development-history:refs/remotes/origin/development-history')
    revision = git('rev-parse', 'origin/development-history').strip()
    state = json.loads(git('show', revision+':docs/work-history/state.json'))
    if repo and state.get('repository') != repo:
        raise ValueError('Repository/history mismatch')
    return state, revision


def event_task(event):
    item = event.get('pull_request') or event.get('issue') or {}
    body = item.get('body') or ''
    ack = ACK_RE.search(body)
    decisions = json.loads(ack.group(1)).get('decisions', []) if ack else []
    # Never rank the acknowledgment's candidate IDs/URLs as part of the task.
    task = item.get('title', '')+'\n'+ACK_RE.sub('', body)
    exclude = 'pr-'+str(item['number']) if event.get('pull_request') else None
    return task.strip(), decisions, exclude


def event_paths(event):
    pr = event.get('pull_request')
    if not pr:
        return []
    base, head = pr['base']['sha'], pr['head']['sha']
    if not all(re.fullmatch(r'[0-9a-f]{40}', sha) for sha in (base, head)):
        raise ValueError('Invalid commit SHA')
    for sha in (base, head):
        try:
            git('cat-file', '-e', sha+'^{commit}')
        except subprocess.CalledProcessError:
            git('fetch', '--quiet', '--no-tags', 'origin', sha)
    return git('diff', '--name-only', '--no-renames', base+'...'+head).splitlines()


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--task'); parser.add_argument('--event'); parser.add_argument('--history')
    parser.add_argument('--paths', nargs='*', default=[]); parser.add_argument('--decisions')
    parser.add_argument('--repo', default=os.environ.get('GITHUB_REPOSITORY'))
    parser.add_argument('--output', default='/tmp/development-preflight')
    parser.add_argument('--no-refresh', action='store_true'); parser.add_argument('--max-age-hours', type=int, default=48)
    args = parser.parse_args()
    output = Path(args.output); output.mkdir(parents=True, exist_ok=True)
    try:
        decisions = json.loads(Path(args.decisions).read_text()) if args.decisions else []
        exclude = None
        if args.event:
            event = json.loads(Path(args.event).read_text())
            task, event_decisions, exclude = event_task(event)
            args.paths = event_paths(event)
            decisions = event_decisions
        else:
            task = args.task or ''
        if len(task.strip()) < 8:
            raise ValueError('Task description too short')
        state, revision = load_history(args.history, args.repo, not args.no_refresh)
        if not REPO_RE.fullmatch(state.get('repository', '')):
            raise ValueError('Invalid repository')
        if args.repo and state['repository'] != args.repo:
            raise ValueError('Repository/history mismatch')
        if args.event and event.get('pull_request'):
            branch = event['pull_request']['head'].get('ref')
            exclude = [exclude]+[c['id'] for c in state['cases'].values()
                                 if c['id'].startswith('branch-') and branch and (c.get('head_ref') or c.get('title')) == branch]
        report = inspect(state, task, args.paths, decisions, exclude, max_age_hours=args.max_age_hours, revision=revision)
    except (OSError, ValueError, KeyError, TypeError, subprocess.SubprocessError):
        # No exception messages: command arguments/event content can contain secrets.
        report = {'status':'INCOMPLETE', 'problems':['履歴または作業内容を取得・検証できない。成功扱いせず復旧後に再照合する']}
        (output/'preflight.json').write_text(json.dumps(report, ensure_ascii=False, indent=2)+'\n')
        (output/'preflight.md').write_text('# 着手前の既存成果自動照合\n\nINCOMPLETE：'+report['problems'][0]+'\n')
        print('INCOMPLETE: '+str(output/'preflight.md'))
        return 2
    (output/'preflight.json').write_text(json.dumps(report, ensure_ascii=False, indent=2)+'\n')
    (output/'preflight.md').write_text(markdown(report))
    print(markdown(report))
    return 2 if report['status']=='INCOMPLETE' else 3 if report['status']=='REVIEW_REQUIRED' else 0


if __name__ == '__main__':
    raise SystemExit(main())

def clean(value, limit=120):

def build(report, inspect_pr=inspect):
    repo = report.get("repository")
    snapshot = report.get("snapshot")
    prs = snapshot.get("prs") if isinstance(snapshot, dict) else None
    if not isinstance(repo, str) or not repo or not isinstance(prs, dict):
        raise ValueError("Missing repository or PR snapshot")
    rows = []
    for number, pr in sorted(prs.items(), key=lambda pair: int(pair[0]) if str(pair[0]).isdigit() else -1):

            continue

            decision = classification.get("decision")

                decision = "hold_invalid"
                reason = "Unknown classifier result"

                decision = "hold_incomplete"
                reason = "PR changed during inventory; recheck required"

                reason = classification.get("reason", "")

            decision = "hold_incomplete"
            reason = "PR classification unavailable: " + type(error).__name__
        rows.append({"number": int(number), "title": clean(pr.get("title", "")),
                     "updated_at": clean(pr.get("updated_at", ""), 35),
                     "decision": decision, "route": ROUTES[decision], "reason": clean(reason, 180),

    return rows

            f" | {row['title']} | {row['route']} | \`{row['decision']}\` | {row['updated_at']} |"
        )
    if not rows:
        lines.append("| — | Open PRなし | — | — | — |")
    lines.extend(["", f"Open PR: {len(rows)}", "",
                  "自動承認・自動マージ・自動通知は行いません。保留も担当ソラが原因を確認してください。"])
    return "\n".join(lines) + "\n"


def main():
    report = json.loads(Path("inventory-output/report.json").read_text(encoding="utf-8"))
    rows = build(report)
    root = Path("inventory-output")
    (root / "pr-approval-routing.json").write_text(
        json.dumps({"repository": report["repository"], "items": rows}, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8")
    markdown = render(report["repository"], rows)
    (root / "pr-approval-routing.md").write_text(markdown, encoding="utf-8")
    print(markdown)
    with open(os.environ.get("GITHUB_STEP_SUMMARY", os.devnull), "a", encoding="utf-8") as output:
        output.write(markdown)


if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""Generate a read-only handoff queue for project maintainers from inventory."""
import json
from pathlib import Path

def render(report):
    repo = report.get("repository", "")
    snapshot = report.get("snapshot")
    prs = snapshot.get("prs") if isinstance(snapshot, dict) else None
    if not repo or not isinstance(prs, dict):
        raise ValueError("Missing repository or PR snapshot")
    lines = ["# 担当ソラへのPR確認待ち一覧", "",
             "この一覧はGitHubのPR状態を記録したものです。安全性や完了を認定しません。",
             "担当ソラは各PRの変更内容・CI・本番影響を確認してください。",
             "ユーザー承認が必要な変更は承認前にmainへ統合しないでください。", "",
             "| PR | タイトル | 状態 | 最終更新 |",
             "| --- | --- | --- | --- |"]
    count = 0
    for number, pr in sorted(prs.items(), key=lambda item: int(item[0]) if str(item[0]).isdigit() else -1):
        if not str(number).isdigit() or not isinstance(pr, dict) or pr.get("state") != "open":
            continue
        title = " ".join(str(pr.get("title", "")).split()).replace("|", "/").replace("[", "(").replace("]", ")")[:120]
        date = str(pr.get("updated_at", ""))[:35]
        lines.append(f"| [#{number}](https://github.com/{repo}/pull/{number}) | {title} | 確認待ち（Draft含む可能性） | {date} |")
        count += 1
    if not count:
        lines.append("| — | 現時点でopen PRなし | — | — |")
    lines.extend(["", f"Open PR: {count}", "",
                  "この一覧は自動承認・自動マージ・担当ソラへのプッシュ通知を行いません。"])
    return "\n".join(lines) + "\n"

def main():
    report = json.loads(Path("inventory-output/report.json").read_text(encoding="utf-8"))
    output = Path("inventory-output/pr-review-handoff.md")
    output.parent.mkdir(parents=True, exist_ok=True)
    result = render(report)
    output.write_text(result, encoding="utf-8")
    print(result)

if __name__ == "__main__":
    main()
