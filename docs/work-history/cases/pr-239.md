+  claimed_recipient: string;
+};
+type Outcome = "SENT" | "FAILED" | "UNKNOWN";
+export type IsolatedPort = {
+  claim: (limit: number) => Promise<IsolatedClaim[]>;
+  finish: (id: number, outcome: Outcome, gmailId: string | null) => Promise<boolean>;
+  mailFor: (claim: IsolatedClaim) => Promise<{ subject: string; body: string }>;
+};
+export async function runIsolatedBatch(
+  port: IsolatedPort,
+  token: string,
+  request: typeof fetch,
+  maxItems = 10
+): Promise<{claimed:number;sent:number;failed:number;unknown:number;stopped:boolean}> {
+  if (!Number.isInteger(maxItems) || maxItems < 1 || maxItems > 10 || !token || !request)
+    throw new Error("ISOLATED_BATCH_NOT_CONFIGURED");
+  // Claim only one at a time: an early stop must not strand the other nine in SENDING.
+  const seen = new Set<number>();
+  let claimedCount=0;
+  let sent=0, failed=0, unknown=0;
+  for (let n=0; n<maxItems; n++) {
+    const claimed = await port.claim(1);
+    if (!Array.isArray(claimed) || claimed.length > 1) throw new Error("ISOLATED_CLAIM_INVALID");
+    if (claimed.length===0) break;
+    const item=claimed[0];
+    claimedCount++;
+    if (!Number.isSafeInteger(item.claimed_id) || item.claimed_id < 1 ||
+        !/^(hiroshima-person|hiroshima-crime|shiga-person):[^\s]{1,120}$/.test(item.claimed_source_id) ||
+        (item.claimed_prefecture === "shiga" ? !item.claimed_source_id.startsWith("shiga-") :
+          !item.claimed_source_id.startsWith("hiroshima-")) ||
+        item.claimed_recipient !== "kishibojim@gmail.com" || seen.has(item.claimed_id))
+      throw new Error("ISOLATED_CLAIM_INVALID");
+    seen.add(item.claimed_id);
+    let result: {status:Outcome;message:string|null} = {status:"UNKNOWN",message:null};
+    try {
+      const mail = await port.mailFor(item);
+      result = await sendIsolatedTestMail(token,mail,request);
+    } catch {
+      // Claim already persisted. Do not retry a potentially attempted send.
+      result = {status:"UNKNOWN",message:null};
+    }
+    let saved=false;
+    try { saved = await port.finish(item.claimed_id,result.status,result.message); }
+    catch { saved=false; }
+    if (!saved) throw new Error("ISOLATED_OUTCOME_NOT_PERSISTED");
+    if (result.status==="SENT") sent++;
+    else if (result.status==="FAILED") failed++;
+    else unknown++;
+    // Fail closed: no further Gmail calls after any non-success.
+    if (result.status!=="SENT") return {claimed:claimedCount,sent,failed,unknown,stopped:true};
+  }
+  return {claimed:claimedCount,sent,failed,unknown,stopped:false};
+}

+/** Issue #238: OAuth refresh transport shared with existing JASPIC Gmail flow.
+ * Does not read credentials, access the database, or send email. The caller
+ * must obtain the refresh token through the existing authenticated vault flow.
+ */
+export async function refreshIsolatedGmailAccessToken(
+  credentials: {clientId:string;clientSecret:string;refreshToken:string},
+  request: typeof fetch
+): Promise<string> {
+  if (!credentials.clientId || !credentials.clientSecret || !credentials.refreshToken || !request)
+    throw new Error("ISOLATED_OAUTH_NOT_CONFIGURED");
+  let response: Response;
+  try {
+    response = await request("https://oauth2.googleapis.com/token", {
+      method: "POST",
+      headers: {"Content-Type":"application/x-www-form-urlencoded"},
+      body: new URLSearchParams({
+        client_id: credentials.clientId,
+        client_secret: credentials.clientSecret,
+        refresh_token: credentials.refreshToken,
+        grant_type: "refresh_token",
+      }),
+      cache: "no-store", redirect: "error", signal: AbortSignal.timeout(10000),
+    });
+  } catch {
+    throw new Error("ISOLATED_OAUTH_REFRESH_UNAVAILABLE");
+  }
+  if (!response.ok) throw new Error("ISOLATED_OAUTH_REFRESH_FAILED");
+  let data: unknown;
+  try { data = await response.json(); }
+  catch { throw new Error("ISOLATED_OAUTH_RESPONSE_INVALID"); }
+  if (!data || typeof data !== "object" ||
+      !("access_token" in data) || typeof data.access_token !== "string" ||
+      data.access_token.length < 8 ||
+      !("token_type" in data) || data.token_type !== "Bearer")
+    throw new Error("ISOLATED_OAUTH_RESPONSE_INVALID");
+  return data.access_token;

+ * No network, credentials, database writes, or mail. */
+export type IsolatedCandidate = { province: "hiroshima" | "shiga"; source_id: string };
+
+export function extractIsolatedCandidates(preview: unknown): IsolatedCandidate[] {
+  if (!preview || typeof preview !== "object") throw new Error("PREVIEW_MISSING");
+  const p = preview as Record<string, unknown>;
+  if (p.mode !== "READ_ONLY" || p.countsComplete !== true ||
+      !Array.isArray(p.provinces) || p.provinces.length !== 2 ||
+      !Array.isArray(p.provinceErrors) || p.provinceErrors.length !== 0) {
+    throw new Error("PREVIEW_INCOMPLETE");
+  }
+  const counts: Record<string, number> = { hiroshima: 31, shiga: 3 };
+  const seen = new Set<string>();
+  const result: IsolatedCandidate[] = [];
+  for (const province of p.provinces) {
+    if (!province || typeof province !== "object") throw new Error("PROVINCE_INVALID");
+    const v = province as Record<string, unknown>;
+    if (typeof v.province !== "string" || !(v.province in counts) ||
+        !Array.isArray(v.keys) || v.keys.length !== counts[v.province] ||
+        v.count !== v.keys.length || seen.has("province:" + v.province)) {
+      throw new Error("PROVINCE_COUNT_MISMATCH");
+    }
+    seen.add("province:" + v.province);
+    for (const key of v.keys) {
+      if (typeof key !== "string" || !/^(hiroshima-person|hiroshima-crime|shiga-person):[^\s]{1,120}$/.test(key) ||
+          (v.province === "shiga" ? !key.startsWith("shiga-person:") : !key.startsWith("hiroshima-")) ||
+          seen.has(key)) throw new Error("CANDIDATE_KEY_INVALID");
+      seen.add(key);
+      result.push({ province: v.province as IsolatedCandidate["province"], source_id: key });
+    }
+  }
+  if (result.length !== 34 || seen.size !== 36 || p.total !== 34) throw new Error("SNAPSHOT_TOTAL_MISMATCH");
