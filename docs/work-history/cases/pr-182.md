+## なぜ前の棚卸しで見つけにくかったのか
+
+従来の棚卸しは、**「どのブランチがある／何が変わった／どのPRがまだ開いている」**を調べるのが得意だった。一方、**「あの実験で具体的に何が分かったか」**は、PRの本文・Actionsログ・別資料を開かないと分からなかった。
+
+今回の自動履歴は、その中間に「**PR本文から作業内容を探す索引**」を追加するもの。**本文冒頭しか自動抜粋しないので、結論と実データは元のPR・ログで確認する**。過去のPR本文に詳細がないと、索引にも出ない。完全な意味理解型の棚卸しだと誤認しない。
+
+## 引継ぎ時の最低手順
+
+1. まずこのファイルの重要実験と、最新の `work-history.md` を検索する。
+2. 類似研究があったら当時のPR・コミット・CIログ・証拠ファイルを開く。
+3. main、本番、実機の差を確認し、**再実装・再測定が本当に必要なときだけ**新規作業を行う。
+4. 新たな重要な結論（見逃し率・負荷限界・プロダクト方針など）は、検索に引っかかる見出しと一次証拠リンクをこの入口にも記録する。

+#!/usr/bin/env python3
+"""Read-only searchable history from existing development inventory snapshot."""
+import json
+import re
+from pathlib import Path
+from urllib.parse import quote
+
+def cell(value):
+    return re.sub(r"\s+", " ", str(value or "")).replace("|", "&#124;").replace("<", "&lt;").replace(">", "&gt;")
+
+def excerpt(body):
+    for original in (body or "").splitlines():
+        line = original.strip().lstrip("-*> ").strip()
+        if line and not line.startswith(("#", "|", "[vc]", "![", "<")) and len(line) >= 12:
+            return cell(line[:180] + ("…" if len(line)>180 else ""))
+    return "説明抜粋なし。PR差分とコメントを確認"
+
+def render(report):
+    snap = report.get("snapshot") or {}
+    prs, branches = snap.get("prs"), snap.get("branches")
+    repo = report.get("repository")
+    if not isinstance(prs, dict) or not isinstance(branches, dict) or not repo:
+        raise ValueError("Incomplete inventory; cannot generate trustworthy index")
+    lines = ["# 作業内容の履歴索引（自動棚卸し）", "",
+             "生成日時 (UTC): " + cell(report.get("generated_at", "不明")), "",
+             "この一覧は過去に何を試したかを探すための索引。説明はPR本文の抜粋であり、検証済みの結論ではない。",
+             "PR状態は実装完了・本番反映を意味しない。PR差分、Actions、開発台帳を確認する。", "",
+             "## 全PRの作業内容", "",
+             "| PR | タイトル | 作業内容（PR本文の抜粋） | 状態 | ブランチ |",
+             "| --- | --- | --- | --- | --- |"]
+    for num,pr in sorted(prs.items(),key=lambda item:int(item[0]) if item[0].isdigit() else -1,reverse=True):
+        if not num.isdigit() or not isinstance(pr,dict): continue
+        link = "[#" + num + "](https://github.com/" + repo + "/pull/" + num + ")"
+        lines.append("| " + " | ".join([link,cell(pr.get("title")),excerpt(pr.get("body")),
+                                            cell(pr.get("state", "unknown")),cell(pr.get("head_ref"))]) + " |")
+    lines += ["", "## 全作業ブランチ（main以外）", "",
+              "ブランチの存在だけで未完了と判断しない。関連PRがないものは作業内容未調査。",
+              "", "| ブランチ | PR履歴 | HEAD SHA |", "| --- | --- | --- |"]
+    for name,sha in sorted(branches.items()):
+        if name == report.get("default_branch"): continue
+        prs_for_branch = sorted((n for n,p in prs.items() if n.isdigit() and isinstance(p,dict) and p.get("head_ref")==name),key=int)
+        refs = ", ".join("[#"+n+"](https://github.com/"+repo+"/pull/"+n+")" for n in prs_for_branch) or "関連PR未確認"
+        lines.append("| ["+cell(name)+"](https://github.com/"+repo+"/tree/"+quote(name,safe="")+") | "+refs+" | "+cell(str(sha or "")[:12])+" |")
+    lines += ["", "## 確認先", "",
+              "- [製品の恒久方針](https://github.com/"+repo+"/blob/main/docs/SAKIKURU_PRODUCT_CHARTER.md)",
+              "- [開発台帳](https://github.com/"+repo+"/blob/main/docs/DEVELOPMENT_LEDGER.md)",
+              "- [全PR](https://github.com/"+repo+"/pulls?q=is%3Apr)", "",
+              "注意: PRの説明や変更ファイルの存在は、その機能が本番動作中である証拠ではない。", ""]
+    return "\n".join(lines)
+
+def main():
+    root=Path("inventory-output")
+    report=json.loads((root/"report.json").read_text(encoding="utf-8"))
+    (root/"work-history.md").write_text(render(report),encoding="utf-8")
+
+if __name__=="__main__": main()

#!/usr/bin/env python3
"""Link branch overlap evidence to existing PRs and explicit index references only."""
import json
import re
from pathlib import Path

def build(report, overlaps, index_text):
    snap = report.get("snapshot")
    if not isinstance(snap, dict) or not isinstance(snap.get("branches"), dict) or not isinstance(snap.get("prs"), dict):
        raise ValueError("Incomplete snapshot")
    if not isinstance(overlaps, dict) or not isinstance(overlaps.get("branches"), dict) or not isinstance(overlaps.get("shared_paths"), dict):
        raise ValueError("Incomplete file evidence")
    rows = re.findall(r"^\|\s*([AM]-\d+)\s*\|([^\n]+)$", index_text, re.M)
    if not rows:
        raise ValueError("Canonical index rows missing")
    cases = []
    for case, fields in rows:
        # Exact branch names only; do not guess from generic titles.
        cases.append((case, fields))
    assignments = []
    for branch, sha in sorted(snap["branches"].items()):
        if branch == report.get("default_branch"):
            continue
        pr_numbers = sorted(
            [int(num) for num, pr in snap["prs"].items()
             if str(num).isdigit() and isinstance(pr, dict) and pr.get("head_ref") == branch])
        explicit = [case for case, fields in cases
                    if branch in re.findall(r"`([^`]+)`", fields)]
        shared = [path for path, names in overlaps["shared_paths"].items()
                  if isinstance(names, list) and branch in names]
        coverage = overlaps["branches"].get(branch, {})
        assignments.append({
            "branch": branch, "head_sha": sha,
            "pr_numbers": pr_numbers,
            "explicit_case_ids": sorted(set(explicit)),
            "shared_changed_paths": shared,
            "file_evidence_complete": bool(isinstance(coverage, dict) and coverage.get("files_complete")),
            "review_required": True
        })
    return assignments

def render(assignments):
    lines = ["# Branch → PR → existing case review handoff", "",
             "This report contains explicit references and changed-path overlap hints only.",
             "A blank case ID means *unmatched*, not missing from main or unfinished.", "",
             "| Branch | PR history | Explicit index case | Shared-file hints | Evidence |",
             "|---|---|---|---|---|"]
    for entry in assignments:

#!/usr/bin/env python3
"""Read-only branch-to-main comparisons. Report evidence, never infer feature completion."""
import json
import hashlib
import os
import urllib.parse
import urllib.request
from pathlib import Path

def compare_rows(report, fetch, evidence=None):
    snapshot = report.get("snapshot") or {}
    branches = snapshot.get("branches") or {}
    default = report.get("default_branch")
    if not isinstance(branches, dict) or default not in branches:
        raise ValueError("Incomplete branch snapshot")
    rows = []
    for name, sha in sorted(branches.items()):
        if name == default:
            continue
        if not isinstance(sha, str) or len(sha) != 40:
            rows.append((name, "inconclusive: missing SHA", "", "", ""))
            continue
        path = "/repos/" + urllib.parse.quote(report["repository"], safe="/") + "/compare/" + urllib.parse.quote(default, safe="") + "..." + urllib.parse.quote(name, safe="")
        try:
            data = fetch(path)
            if not isinstance(data, dict):
                raise ValueError("Unexpected API result")
            status = data.get("status", "unknown")
            ahead = data.get("ahead_by", "?")
            behind = data.get("behind_by", "?")
            files = data.get("files", [])
            if not isinstance(files, list):
                raise ValueError("Missing files list")
            # GitHub compare returns a capped file list. Do not claim coverage at or near the cap.
            coverage = "partial/unknown" if len(files) >= 300 or ahead == "?" else "API-reported"
            if evidence is not None:
                evidence[name] = {
                    "head_sha": sha,
                    "status": status,
                    "files_complete": len(files) < 300 and ahead != "?",
                    "paths": sorted({item["filename"] for item in files
                                     if isinstance(item, dict) and isinstance(item.get("filename"), str)}),
                    "patch_sha256": {item["filename"]: hashlib.sha256(item["patch"].encode("utf-8")).hexdigest()
                                     for item in files if isinstance(item, dict)
                                     and isinstance(item.get("filename"), str)
                                     and isinstance(item.get("patch"), str)
                                     and item["patch"]},
                    "patch_absence": [item["filename"] for item in files if isinstance(item, dict)
                                      and isinstance(item.get("filename"), str)
                                      and not isinstance(item.get("patch"), str)]
                }
            sample = ", ".join(str(f.get("filename", "")) for f in files[:8] if isinstance(f, dict))
            rows.append((name, str(status), str(ahead), str(behind), coverage + ": " + sample))
        except (ValueError, KeyError, OSError) as error:
            rows.append((name, "inconclusive: " + type(error).__name__, "", "", ""))
    return rows

def find_shared_paths(evidence):
    """Potential review leads, not a statement of duplicated implementations."""
    if not isinstance(evidence, dict):
        raise ValueError("Missing evidence")
    by_file = {}
    for branch, info in evidence.items():
        if not isinstance(info, dict):
            continue
        for path in info.get("paths", []):
            # Documentation and CI infrastructure overlap frequently; ignore those paths.
            if path.startswith(("docs/", ".github/", "tests/")) or path.endswith((".md", ".lock")):
                continue
            by_file.setdefault(path, []).append(branch)
    return {path: sorted(names) for path, names in sorted(by_file.items()) if len(names) > 1}


def find_equal_patches(evidence):
    """Exact patch equality only. Hashes are evidence, not a duplicate-feature decision."""
    matches = {}
    groups = {}
    for branch, info in sorted(evidence.items()):
        if not isinstance(info, dict):
            continue
        hashes = info.get("patch_sha256", {})
        if not isinstance(hashes, dict):
            continue
        for path, digest in hashes.items():
            if not isinstance(digest, str) or len(digest) != 64:
                continue
            if path.startswith(("docs/", ".github/", "tests/")) or path.endswith((".md", ".lock")):
                continue
            groups.setdefault((path, digest), []).append(branch)
    for (path, digest), branches in sorted(groups.items()):
        if len(branches) >= 2:
            matches.setdefault(path, []).append({"patch_sha256": digest, "branches": sorted(branches)})
    return matches


def render_equal_patches(matches):
    lines = ["# Identical comparison patches (review only)", "",
             "Matching GitHub compare patch text SHA-256 indicates identical API-returned diff hunks",
             "for one file at the time of this snapshot. This is NOT proof of duplicate features,",
             "commits, or merge safety. Missing/truncated patches cannot be assessed.", "",
             "| Changed path | Matching branch heads | Patch SHA-256 |", "|---|---|---|"]
    for path, groups in sorted(matches.items()):
        for group in groups:
            safe_path = path.replace("|", "&#124;").replace("\\n", " ")
            names = ", ".join(branch.replace("|", "&#124;").replace("\\n", " ") for branch in group["branches"])
            lines.append("| " + safe_path + " | " + names + " | " + group["patch_sha256"] + " |")
    if not matches:
        lines.append("| None observed in available patches | — | — |")
    return "\\n".join(lines) + "\\n"


def render_shared_paths(overlaps, evidence):
    lines = ["# Shared changed-file candidates (review only)", "",
             "Multiple branches changing the same path does NOT establish duplicate functionality.",
             "Check actual patches, merged PR history, and the development ledger.", "",
             "| File | Branches | Evidence coverage |", "|---|---|---|"]
    for path, names in overlaps.items():
        coverage = "partial/unknown" if any(
            not evidence[name].get("files_complete", False) for name in names) else "API-reported"
        safe_path = path.replace("|", "&#124;").replace("\n", " ")
        safe_names = ", ".join(name.replace("|", "&#124;").replace("\n", " ") for name in names)
        lines.append(f"| {safe_path} | {safe_names} | {coverage} |")
    if not overlaps:
        lines.extend(["", "No shared changed paths found in the available API evidence; "
                      "this does not prove absence of duplicated work."])
    return "\n".join(lines) + "\n"


def render(rows):
    lines = ["# Branch-to-main evidence (read-only)", "",
             "This is GitHub compare metadata, not proof that a feature is missing, complete, or safe to merge.",
             "Review full diff, closed PRs, and ledger before taking action.", "",
             "| Branch | Compare status | Ahead | Behind | File evidence |",
             "|---|---|---:|---:|---|"]
    for fields in rows:
        escaped = [str(value).replace("|", "&#124;").replace("\n", " ") for value in fields]
        lines.append("| " + " | ".join(escaped) + " |")
    return "\n".join(lines) + "\n"

def main():
    report = json.loads(Path("inventory-output/report.json").read_text(encoding="utf-8"))
    repo = report["repository"]
    token = os.environ["GITHUB_TOKEN"]
    def fetch(path):
        req = urllib.request.Request("https://api.github.com" + path, headers={
            "Authorization": "Bearer " + token,
            "Accept": "application/vnd.github+json",
            "X-GitHub-Api-Version": "2022-11-28",
            "User-Agent": "branch-to-main-inventory"})
        with urllib.request.urlopen(req, timeout=20) as response:
            return json.load(response)
    evidence = {}
    rows = compare_rows(report, fetch, evidence=evidence)
    Path("inventory-output/branch-comparison.md").write_text(render(rows), encoding="utf-8")
    overlaps = find_shared_paths(evidence)
    equal_patches = find_equal_patches(evidence)
    Path("inventory-output/branch-file-evidence.json").write_text(

        encoding="utf-8")
    Path("inventory-output/shared-file-review.md").write_text(
        render_shared_paths(overlaps, evidence), encoding="utf-8")
    Path("inventory-output/equal-patch-review.md").write_text(

    print("Recorded comparisons for " + str(len(rows)) + " branches in " + repo)

if __name__ == "__main__":

"""Prioritize human review of branch evidence without deciding implementation status."""
import json
from pathlib import Path

def prioritize(handoff, evidence):
    items = handoff.get("assignments")
    shared = evidence.get("shared_paths")
    equal = evidence.get("equal_patches")
    if not isinstance(items, list) or not isinstance(shared, dict) or not isinstance(equal, dict):
        raise ValueError("Incomplete handoff or patch evidence")
    result = []
    for row in items:
        if not isinstance(row, dict) or not isinstance(row.get("branch"), str):
            raise ValueError("Malformed handoff entry")
        branch = row["branch"]
        exact_paths = sorted(path for path, groups in equal.items()
                             if isinstance(groups, list) and any(
                                 isinstance(group, dict) and branch in group.get("branches", [])
                                 for group in groups))
        reasons = []
        if exact_paths:
            reasons.append("identical_api_patch_hints")
        if not row.get("file_evidence_complete"):
            reasons.append("file_evidence_incomplete")
        if not row.get("explicit_case_ids"):
            reasons.append("no_explicit_case_reference")
        if not row.get("pr_numbers"):
            reasons.append("no_historical_pr_reference")
        overlap = sorted(path for path, branches in shared.items()
                         if isinstance(branches, list) and branch in branches)
        if overlap:
            reasons.append("shared_file_paths")
        priority = ("P1_review" if exact_paths or not row.get("file_evidence_complete")
                    else "P2_review" if overlap or not row.get("explicit_case_ids") or not row.get("pr_numbers")
                    else "P3_review")
        result.append({
            "branch": branch, "priority": priority, "reasons": reasons,
            "explicit_case_ids": row.get("explicit_case_ids", []),
            "historical_pr_numbers": row.get("pr_numbers", []),
            "equal_patch_paths": exact_paths[:10],
            "shared_file_paths": overlap[:10],
            "next_action": "Review actual main diff, closed/merged PRs, and existing case notes",
            "auto_applied": False
        })
    return sorted(result, key=lambda x: (x["priority"], x["branch"]))

            ", ".join(item["reasons"]) or "routine review",

if __name__ == "__main__":

#!/usr/bin/env python3
"""Read-only GitHub development inventory. No branch/PR/Issue mutations."""
import datetime as dt
import json
import os
import sys
import urllib.error
import urllib.parse
import urllib.request

REPO = os.environ["GITHUB_REPOSITORY"]
TOKEN = os.environ["GITHUB_TOKEN"]
NOW = dt.datetime.now(dt.timezone.utc)
STALE_DAYS = int(os.environ.get("STALE_DAYS", "14"))
API = "https://api.github.com"

def get(path):
    req = urllib.request.Request(API + path, headers={
        "Authorization": "Bearer " + TOKEN,
        "Accept": "application/vnd.github+json",
        "X-GitHub-Api-Version": "2022-11-28",
        "User-Agent": "development-inventory-readonly",
    })
    with urllib.request.urlopen(req, timeout=30) as response:
        return json.load(response)

def pages(path):
    result = []
    page = 1
    while True:
        separator = "&" if "?" in path else "?"
        batch = get(path + separator + "per_page=100&page=" + str(page))
        if not isinstance(batch, list):
            raise RuntimeError("Unexpected API response")
        result.extend(batch)
        if len(batch) < 100:
            return result
        page += 1

def age(timestamp):
    if not timestamp:
        return None
    updated = dt.datetime.fromisoformat(timestamp.replace("Z", "+00:00"))
    return (NOW - updated).days

def main():
    repo = urllib.parse.quote(REPO, safe="/")
    metadata = get("/repos/" + repo)
    default = metadata["default_branch"]
    branches = pages("/repos/" + repo + "/branches")
    all_prs = pages("/repos/" + repo + "/pulls?state=all")
    prs = [p for p in all_prs if p["state"] == "open"]
    issues = [item for item in pages("/repos/" + repo + "/issues?state=open") if "pull_request" not in item]
    pr_heads = {p["head"]["ref"] for p in prs if p["head"]["repo"] and p["head"]["repo"]["full_name"] == REPO}
    historical_heads = {p["head"]["ref"] for p in all_prs if p["head"]["repo"] and p["head"]["repo"]["full_name"] == REPO}
    findings = []
    for branch in branches:
        name = branch["name"]
        if name == default:
            continue
        # No-PR does NOT mean unfinished. Report only as a review candidate.
        if name not in pr_heads:
            findings.append({"type": "branch_without_open_pr", "name": name,
                             "historical_pr_found": name in historical_heads,
                             "url": "https://github.com/" + REPO + "/tree/" + urllib.parse.quote(name, safe="")})
    stale_prs = [{"number": p["number"], "title": p["title"], "days": age(p["updated_at"]),
                  "url": p["html_url"]} for p in prs if age(p["updated_at"]) is not None and age(p["updated_at"]) >= STALE_DAYS]
    stale_issues = [{"number": i["number"], "title": i["title"], "days": age(i["updated_at"]),
                     "url": i["html_url"]} for i in issues if age(i["updated_at"]) is not None and age(i["updated_at"]) >= STALE_DAYS]
    report = {"repository": REPO, "generated_at": NOW.isoformat(), "default_branch": default,
              "branch_count": len(branches), "open_pr_count": len(prs),
              "open_issue_count": len(issues), "historical_pr_count": len(all_prs), "review_candidates": findings,
              "stale_prs": stale_prs, "stale_issues": stale_issues,
              "caveat": "Candidates are NOT proof of unfinished work. Compare main, merged/closed PRs, and canonical ledger before action."}
    # Stable snapshot for comparison across successful scheduled/manual runs.
    report["snapshot"] = {
        "branches": {b["name"]: b.get("commit", {}).get("sha") for b in branches},
        "prs": {str(p["number"]): {"state": ("merged" if p.get("merged_at") else p["state"]),
                                  "head": p["head"].get("sha"), "updated_at": p["updated_at"],
                                  "title": p.get("title", ""), "body": (p.get("body") or "")[:2000],
                                  "head_ref": p["head"].get("ref", "")}
                for p in all_prs},
        "issues": {str(i["number"]): {"state": "open", "updated_at": i["updated_at"]} for i in issues},
    }
    os.makedirs("inventory-output", exist_ok=True)
    with open("inventory-output/report.json", "w", encoding="utf-8") as f:
        json.dump(report, f, ensure_ascii=False, indent=2)
    summary = ["# Development inventory (read-only)", "",
               "Repository: " + REPO, "Generated (UTC): " + NOW.isoformat(), "",
               "| Metric | Count |", "|---|---:|",
               "| Branches including default | " + str(len(branches)) + " |",
               "| Open PRs | " + str(len(prs)) + " |",
               "| Open issues | " + str(len(issues)) + " |",
               "| Branches without open PR (review only) | " + str(len(findings)) + " |",
               "| Stale PRs (>= " + str(STALE_DAYS) + " days) | " + str(len(stale_prs)) + " |",
               "| Stale issues | " + str(len(stale_issues)) + " |", "",
               "**Do not mark branches unfinished or delete them based on this report.**",
               "Consult the project's canonical ledger, closed/merged PRs and main code first.", "",
               "## Stale PRs"]
    summary.extend("- [#" + str(p["number"]) + "](" + p["url"] + ") (" + str(p["days"]) + "d): " + p["title"].replace("\n", " ") for p in stale_prs)
    summary.append("\n## Stale issues")
    summary.extend("- [#" + str(i["number"]) + "](" + i["url"] + ") (" + str(i["days"]) + "d): " + i["title"].replace("\n", " ") for i in stale_issues)
    summary.append("\nBranch candidates are available in report.json artifact.")
    text = "\n".join(summary) + "\n"
    with open("inventory-output/summary.md", "w", encoding="utf-8") as f:
        f.write(text)
    with open(os.environ.get("GITHUB_STEP_SUMMARY", "/dev/null"), "a", encoding="utf-8") as f:
        f.write(text)
    print(text)

if __name__ == "__main__":
    try:
        main()

import urllib.error
import urllib.request
import urllib.error
import zipfile
from pathlib import Path

API = "https://api.github.com"
REPO = os.environ.get("GITHUB_REPOSITORY", "")
TOKEN = os.environ.get("GITHUB_TOKEN", "")
CURRENT_RUN = str(os.environ.get("GITHUB_RUN_ID", ""))

class SafeRedirect(urllib.request.HTTPRedirectHandler):
    """Do not forward GitHub bearer tokens to external artifact storage."""
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        redirected = super().redirect_request(req, fp, code, msg, headers, newurl)
        if redirected is not None and urllib.parse.urlsplit(newurl).hostname != urllib.parse.urlsplit(req.full_url).hostname:
            redirected.remove_header("Authorization")
            redirected.unredirected_hdrs.pop("Authorization", None)
        return redirected

OPENER = urllib.request.build_opener(SafeRedirect)

def api(path, binary=False):
    req = urllib.request.Request(API + path, headers={
        "Authorization": "Bearer " + TOKEN,
        "Accept": "application/vnd.github+json",
        "X-GitHub-Api-Version": "2022-11-28",
        "User-Agent": "development-inventory-delta-readonly",
    })
    try:
        response = OPENER.open(req, timeout=30)
    except urllib.error.HTTPError as error:
        # Expose the failed endpoint *path*, never signed query parameters or tokens.
        location = urllib.parse.urlsplit(error.url)
        raise RuntimeError("GitHub artifact lookup HTTP " + str(error.code) +
                           " at " + location.hostname + location.path) from error
    with response:
        return response.read() if binary else json.load(response)

def previous_snapshot():
    repo = urllib.parse.quote(REPO, safe="/")
    runs = api("/repos/" + repo + "/actions/workflows/development-inventory.yml/runs?status=success&per_page=30")
    for run in runs.get("workflow_runs", []):
        if str(run["id"]) == CURRENT_RUN or run["event"] not in ("schedule", "workflow_dispatch"):
            continue
        artifacts = api("/repos/" + repo + "/actions/runs/" + str(run["id"]) + "/artifacts?per_page=100")
        for item in artifacts.get("artifacts", []):
            if item["name"] != "development-inventory" or item.get("expired"):
                continue
            archive = api("/repos/" + repo + "/actions/artifacts/" + str(item["id"]) + "/zip", binary=True)
            with zipfile.ZipFile(io.BytesIO(archive)) as z:
                with z.open("report.json") as f:
                    previous = json.load(f)
            if previous.get("repository") == REPO and "snapshot" in previous:
                return previous, run["id"]
    return None, None

def compare(previous, current):
    changes = []
    for group in ("branches", "prs", "issues"):
        before = previous.get("snapshot", {}).get(group, {})
        after = current["snapshot"][group]
        for key in sorted(set(before) | set(after)):
            if key not in before:
                kind = "added"
            elif key not in after:
                kind = "absent_from_snapshot"
            elif before[key] != after[key]:
                kind = "changed"
            else:
                continue
            changes.append({"type": group, "id": key, "change": kind,
                            "before": before.get(key), "after": after.get(key)})
    return changes


def classify_candidates(changes, index_text):
    """Conservative review-only matching; PR files are evidence, never proof."""
    import re
    rows = re.findall(r"^\|\s*([AM]-\d+)\s*\|\s*([^|]+)\|", index_text, re.M)
    suggestions = []
    for change in changes:
        identifier = str(change["id"])
        details = change.get("after") or change.get("before") or {}
        # Branch snapshots contain SHA strings, while PR snapshots contain dicts.
        # Never treat a branch SHA as PR metadata.
        if not isinstance(details, dict):
            details = {}
        metadata = " ".join(str(details.get(k, "")) for k in ("title", "body", "head_ref"))
        filenames = []
        file_status = "not_applicable"
        if change["type"] == "prs" and identifier.isdigit() and REPO:
            try:
                path = "/repos/" + urllib.parse.quote(REPO, safe="/") + "/pulls/" + identifier + "/files?per_page=100"
                files = api(path)
                filenames = [item["filename"] for item in files if "filename" in item]
                file_status = "partial_first_100" if len(files) == 100 else "complete"
            except (RuntimeError, urllib.error.URLError, ValueError, KeyError):
                file_status = "unavailable"
        text = (identifier + " " + metadata + " " + " ".join(filenames)).lower()
        tokens = set(re.findall(r"[a-z0-9]{4,}", text.replace("_", "-")))
        tokens.update(re.findall(r"[\u3040-\u30ff\u4e00-\u9fff]{2,}", text))
        matches = []
        for case_id, title in rows:
            title_tokens = set(re.findall(r"[a-z0-9]{4,}", title.lower()))
            title_tokens.update(re.findall(r"[\u3040-\u30ff\u4e00-\u9fff]{2,}", title))
            if tokens & title_tokens:
                matches.append(case_id)
        # Generic infrastructure PRs are not forced into product work items.
        infrastructure = any(x in text for x in ("development_inventory", "development-inventory", "inventory-triage"))
        if infrastructure:
            matches = []
        suggestions.append({
            "source_type": change["type"], "source_id": identifier,
            "change": change["change"], "candidate_case_ids": matches,
            "classification": ("automation_maintenance" if infrastructure else
                               "possible_existing" if matches else "needs_review"),
            "ledger_action": "review_only", "proposed_status": None,
            "evidence": "PR metadata and changed file names; unverified token hints",
            "file_evidence": {"status": file_status, "paths": filenames[:100]}
        })
    return suggestions

def write_triage(output, result):
    index_path = Path("docs/DEVELOPMENT_OPERATIONS_INDEX.md")
    index_text = index_path.read_text(encoding="utf-8") if index_path.exists() else ""
    suggestions = classify_candidates(result["changes"], index_text) if result["status"] == "compared" else []
    payload = {"status": result["status"], "previous_run_id": result["previous_run_id"],
               "suggestions": suggestions, "auto_applied": False}
    (output / "triage-proposals.json").write_text(
        json.dumps(payload, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    lines = ["# 台帳更新案（承認待ち・自動反映なし）", "",
             "比較状態: " + result["status"], "",
             "候補は照合用のヒントです。完了・未完了・削除可否は自動判定しません。", ""]
    if result["status"] != "compared":
        lines.append("比較未完了のため更新案を作成していません。")
    for item in suggestions:
        linked = ", ".join(item["candidate_case_ids"]) or "未特定"
        lines.extend(["- " + item["source_type"] + " " + item["source_id"] +
                      " (" + item["change"] + ")",
                      "  - 関連候補: " + linked + "（未検証）",
                      "  - 台帳変更案: 照合後に人が判断（状態変更なし）"])
    (output / "ledger-review-proposals.md").write_text("\n".join(lines) + "\n", encoding="utf-8")

def main():
    output = Path("inventory-output")
    current = json.loads((output / "report.json").read_text(encoding="utf-8"))
    error_note = None
    try:
        previous, run_id = previous_snapshot()
    except (RuntimeError, urllib.error.HTTPError, urllib.error.URLError, zipfile.BadZipFile, KeyError, ValueError) as error:
        # Never discard today's complete inventory because a prior artifact cannot be read.
        # Explicitly report the comparison failure; do NOT claim zero changes.
        previous, run_id = None, None
        error_note = "Previous report unavailable: " + type(error).__name__
        if isinstance(error, urllib.error.HTTPError):
            error_note += " (HTTP " + str(error.code) + ")"
        if isinstance(error, RuntimeError):
            error_note += " (" + str(error) + ")"
    if error_note:
        result = {"status": "comparison_unavailable", "previous_run_id": None, "changes": [],
                  "note": error_note + "; delta NOT checked. Current inventory saved."}
    elif previous is None:
        result = {"status": "baseline_only", "previous_run_id": None, "changes": [],
                  "note": "No prior compatible successful artifact; do not infer no changes."}
    else:
        result = {"status": "compared", "previous_run_id": run_id,
                  "changes": compare(previous, current),
                  "note": "Candidates only. No automatic ledger status or completion decisions."}
    (output / "delta.json").write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    lines = ["# Development inventory change candidates", "",
             "Status: " + result["status"],
             "Previous run: " + str(run_id or "none"),
             "Candidates: " + str(len(result["changes"])), "",
             result["note"], "",
             "**Read-only suggestions; verify main, PR history and ledger before updating.**", ""]
    for c in result["changes"]:
        lines.append("- " + c["type"] + " `" + c["id"].replace("`", "") + "`: " + c["change"])
    (output / "ledger-update-candidates.md").write_text("\n".join(lines) + "\n", encoding="utf-8")
    write_triage(output, result)
    print("\n".join(lines))

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""Generate a deterministic, review-only GitHub facts section from inventory report."""
import argparse
import json
from pathlib import Path

START = "<!-- INVENTORY_FACTS_START -->"
END = "<!-- INVENTORY_FACTS_END -->"

def render(report):
    prs = report.get("snapshot", {}).get("prs", {})
    lines = ["## GitHub自動取得の事実（確認用）", "",
             "> この欄はPRメタデータの記録であり、開発完了・本番反映の認定ではありません。", "",
             "| PR | タイトル | 状態 | head SHA | 更新日時 |",
             "| --- | --- | --- | --- | --- |"]
    for number, pr in sorted(prs.items(), key=lambda item: int(item[0]) if str(item[0]).isdigit() else -1):
        if not str(number).isdigit() or not isinstance(pr, dict):
            continue
        if str(pr.get("head_ref", "")).startswith("automation/ledger-facts-"):
            continue
        state = str(pr.get("state", "unknown")).replace("|", "/").replace("\n", " ")
        sha = str(pr.get("head", ""))[:40]
        updated = str(pr.get("updated_at", ""))[:35]
        title = " ".join(str(pr.get("title", "")).split()).replace("|", "/")[:120]
        lines.append(f"| #{number} | {title} | {state} | `{sha}` | {updated} |")
    return "\n".join(lines) + "\n"

def update(ledger, report):
    section = START + "\n" + render(report) + END
    if START in ledger or END in ledger:
        if ledger.count(START) != 1 or ledger.count(END) != 1 or ledger.index(START) > ledger.index(END):
            raise ValueError("Invalid managed markers; refuse to change ledger")
        return ledger[:ledger.index(START)] + section + ledger[ledger.index(END) + len(END):]
    return ledger.rstrip() + "\n\n" + section + "\n"

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--report", default="inventory-output/report.json")
    parser.add_argument("--ledger", default="docs/DEVELOPMENT_LEDGER.md")
    parser.add_argument("--output", default="inventory-output/ledger-proposed.md")
    args = parser.parse_args()
    report = json.loads(Path(args.report).read_text(encoding="utf-8"))
    if not isinstance(report.get("snapshot", {}).get("prs"), dict):
        raise ValueError("PR snapshot missing; refuse to propose ledger update")
    proposed = update(Path(args.ledger).read_text(encoding="utf-8"), report)
    target = Path(args.output)
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(proposed, encoding="utf-8")

if __name__ == "__main__":
    main()

Read-only. Decisions are advisory, never proof of review or approval.

ROUTES = {

def clean(value, limit=120):
    return " ".join(str(value).split()).replace("|", "/").replace("[", "(").replace("]", ")")[:limit]


def build(report, inspect_pr=inspect):
    repo = report.get("repository")
    snapshot = report.get("snapshot")
    prs = snapshot.get("prs") if isinstance(snapshot, dict) else None
    if not isinstance(repo, str) or not repo or not isinstance(prs, dict):
        raise ValueError("Missing repository or PR snapshot")
    rows = []
    for number, pr in sorted(prs.items(), key=lambda pair: int(pair[0]) if str(pair[0]).isdigit() else -1):
        if not str(number).isdigit() or not isinstance(pr, dict) or pr.get("state") != "open":
            continue
        try:
            classification = inspect_pr(repo, int(number))
            decision = classification.get("decision")
            if decision not in ROUTES:
                decision = "hold_invalid"
                reason = "Unknown classifier result"
            elif classification.get("head_sha") != pr.get("head"):
                decision = "hold_incomplete"
                reason = "PR changed during inventory; recheck required"
            else:
                reason = classification.get("reason", "")
        except (urllib.error.URLError, ValueError, KeyError, TypeError, TimeoutError) as error:
            decision = "hold_incomplete"
            reason = "PR classification unavailable: " + type(error).__name__
        rows.append({"number": int(number), "title": clean(pr.get("title", "")),
                     "updated_at": clean(pr.get("updated_at", ""), 35),
                     "decision": decision, "route": ROUTES[decision], "reason": clean(reason, 180),
                     "head_sha": pr.get("head")})
    return rows

#!/usr/bin/env python3
"""Conservative PR promotion classifier. Never merges, approves, or deploys."""
import argparse
import json
from pathlib import PurePosixPath

HIGH_RISK_PREFIXES = (
    "src/", "app/", "pages/", "api/", "supabase/", "database/",
    "migrations/", "functions/", "lib/", "server/", "public/",
)
HIGH_RISK_EXACT = {
    "package.json", "package-lock.json", "pnpm-lock.yaml",
    "vercel.json", "next.config.js", "next.config.ts",
    ".env", ".env.production",
}
POLICY_PREFIXES = (".github/", "scripts/",)
FACTS_LEDGER = "docs/DEVELOPMENT_LEDGER.md"

def classify(paths, *, draft=False, complete=True):
    """Fail closed: file names are insufficient proof of safe content."""
    paths = list(paths)
    if draft:
        return {"decision": "hold_draft", "reason": "Draft PR; never promote"}
    if not complete or not paths:
        return {"decision": "hold_incomplete", "reason": "Missing/incomplete file evidence"}
    for path in paths:
        if not isinstance(path, str) or not path or path.startswith("/") or "\\" in path:
            return {"decision": "hold_invalid", "reason": "Invalid changed path"}
        if ".." in PurePosixPath(path).parts:
            return {"decision": "hold_invalid", "reason": "Traversal in changed path"}
    if any(p.startswith(HIGH_RISK_PREFIXES) or p in HIGH_RISK_EXACT
           or p.startswith(".env") for p in paths):
        return {"decision": "owner_approval_required", "reason": "Production or app behavior may change"}
    if any(p.startswith(POLICY_PREFIXES) for p in paths):
        return {"decision": "maintainer_review_required", "reason": "Automation or policy changes cannot self-approve"}
    if FACTS_LEDGER in paths:
        return {"decision": "maintainer_review_required", "reason": "Ledger changes require managed-section diff verification"}
    return {"decision": "maintainer_review_required", "reason": "No content-level allow rule has been verified"}

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--files-json", required=True, help="JSON list of all changed paths")
    parser.add_argument("--draft", action="store_true")
    parser.add_argument("--incomplete", action="store_true")
    args = parser.parse_args()
    with open(args.files_json, encoding="utf-8") as f:
        paths = json.load(f)
    if not isinstance(paths, list):
        raise ValueError("Expected JSON list of paths")
    print(json.dumps(classify(paths, draft=args.draft, complete=not args.incomplete), ensure_ascii=False))

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""Classify a PR using trusted main-branch policy and GitHub API metadata. Read-only."""
import json
import os
import sys
import urllib.error
import urllib.parse
import urllib.request
from pathlib import Path

try:
    from scripts.pr_promotion_policy import classify
except ModuleNotFoundError:
    from pr_promotion_policy import classify


def fetch_json(path):
    req = urllib.request.Request(
        "https://api.github.com" + path,
        headers={
            "Authorization": "Bearer " + os.environ["GITHUB_TOKEN"],
            "Accept": "application/vnd.github+json",
            "X-GitHub-Api-Version": "2022-11-28",
            "User-Agent": "promotion-review-only",
        },
    )
    with urllib.request.urlopen(req, timeout=30) as response:
        return json.load(response)


def inspect(repo, number):
    root = "/repos/" + urllib.parse.quote(repo, safe="/")
    pr = fetch_json(root + "/pulls/" + str(number))
    if pr.get("base", {}).get("ref") != "main":
        return {"decision": "hold_invalid", "reason": "Target branch is not main"}
    paths = []
    complete = True
    for page in range(1, 12):
        batch = fetch_json(root + "/pulls/" + str(number) + "/files?per_page=100&page=" + str(page))
        if not isinstance(batch, list):
            complete = False
            break
        if any(not isinstance(item, dict) or not isinstance(item.get("filename"), str) for item in batch):
            complete = False
            break
        paths.extend(item["filename"] for item in batch)
        if len(batch) < 100:
            break
    else:
        complete = False
    if pr.get("changed_files") != len(paths):
        complete = False
    result = classify(paths, draft=bool(pr.get("draft", True)), complete=complete)
    return {
        **result, "pr": number, "head_sha": pr.get("head", {}).get("sha"),
        "files_seen": len(paths), "files_complete": complete,
        "auto_merged": False, "approval_granted": False,
    }


def main():
    repo = os.environ["GITHUB_REPOSITORY"]
    number = int(os.environ["PR_NUMBER"])
    try:
        result = inspect(repo, number)
    except (urllib.error.URLError, ValueError, KeyError, TypeError) as error:
        result = {
            "decision": "hold_incomplete", "reason": "GitHub metadata unavailable: " + type(error).__name__,
            "pr": number, "auto_merged": False, "approval_granted": False,
        }
    output = Path("inventory-output/pr-promotion-review.json")
    output.parent.mkdir(parents=True, exist_ok=True)
    output.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    summary = ("### PR promotion review (read-only)\n"
               "- PR: #" + str(number) + "\n"
               "- Decision: **" + result["decision"] + "**\n"
               "- Reason: " + result["reason"] + "\n"
               "- No approval, merge, or deployment performed.\n")
    print(summary)
    with open(os.environ.get("GITHUB_STEP_SUMMARY", os.devnull), "a", encoding="utf-8") as f:
        f.write(summary)

def may_skip(previous):
    if not previous or len(previous) != 40 or any(c not in "0123456789abcdefABCDEF" for c in previous):

    try:
        result = subprocess.run(["git", "diff", "--name-only", previous, "HEAD", "--"], capture_output=True, text=True, timeout=20)
    except (OSError, subprocess.TimeoutExpired):
        return False
    paths = result.stdout.splitlines()
    return result.returncode == 0 and bool(paths) and all(p in MANAGEMENT_FILES for p in paths)
