#!/usr/bin/env python3
"""Read-only GitHub development inventory. No branch/PR/Issue mutations."""
import datetime as dt
import json
import os
import sys
import urllib.error
import urllib.parse
import urllib.request

REPO = os.environ["GITHUB_REPOSITORY"]
TOKEN = os.environ["GITHUB_TOKEN"]
NOW = dt.datetime.now(dt.timezone.utc)
STALE_DAYS = int(os.environ.get("STALE_DAYS", "14"))
API = "https://api.github.com"

def get(path):
    req = urllib.request.Request(API + path, headers={
        "Authorization": "Bearer " + TOKEN,
        "Accept": "application/vnd.github+json",
        "X-GitHub-Api-Version": "2022-11-28",
        "User-Agent": "development-inventory-readonly",
    })
    with urllib.request.urlopen(req, timeout=30) as response:
        return json.load(response)

def pages(path):
    result = []
    page = 1
    while True:
        separator = "&" if "?" in path else "?"
        batch = get(path + separator + "per_page=100&page=" + str(page))
        if not isinstance(batch, list):
            raise RuntimeError("Unexpected API response")
        result.extend(batch)
        if len(batch) < 100:
            return result
        page += 1

def age(timestamp):
    if not timestamp:
        return None
    updated = dt.datetime.fromisoformat(timestamp.replace("Z", "+00:00"))
    return (NOW - updated).days

def main():
    repo = urllib.parse.quote(REPO, safe="/")
    metadata = get("/repos/" + repo)
    default = metadata["default_branch"]
    branches = pages("/repos/" + repo + "/branches")
    all_prs = pages("/repos/" + repo + "/pulls?state=all")
    prs = [p for p in all_prs if p["state"] == "open"]
    issues = [item for item in pages("/repos/" + repo + "/issues?state=open") if "pull_request" not in item]
    pr_heads = {p["head"]["ref"] for p in prs if p["head"]["repo"] and p["head"]["repo"]["full_name"] == REPO}
    historical_heads = {p["head"]["ref"] for p in all_prs if p["head"]["repo"] and p["head"]["repo"]["full_name"] == REPO}
    findings = []
    for branch in branches:
        name = branch["name"]
        if name == default:
            continue
        # No-PR does NOT mean unfinished. Report only as a review candidate.
        if name not in pr_heads:
            findings.append({"type": "branch_without_open_pr", "name": name,
                             "historical_pr_found": name in historical_heads,
                             "url": "https://github.com/" + REPO + "/tree/" + urllib.parse.quote(name, safe="")})
    stale_prs = [{"number": p["number"], "title": p["title"], "days": age(p["updated_at"]),
                  "url": p["html_url"]} for p in prs if age(p["updated_at"]) is not None and age(p["updated_at"]) >= STALE_DAYS]
    stale_issues = [{"number": i["number"], "title": i["title"], "days": age(i["updated_at"]),
                     "url": i["html_url"]} for i in issues if age(i["updated_at"]) is not None and age(i["updated_at"]) >= STALE_DAYS]
    report = {"repository": REPO, "generated_at": NOW.isoformat(), "default_branch": default,
              "branch_count": len(branches), "open_pr_count": len(prs),
              "open_issue_count": len(issues), "historical_pr_count": len(all_prs), "review_candidates": findings,
              "stale_prs": stale_prs, "stale_issues": stale_issues,
              "caveat": "Candidates are NOT proof of unfinished work. Compare main, merged/closed PRs, and canonical ledger before action."}
    os.makedirs("inventory-output", exist_ok=True)
    with open("inventory-output/report.json", "w", encoding="utf-8") as f:
        json.dump(report, f, ensure_ascii=False, indent=2)
    summary = ["# Development inventory (read-only)", "",
               "Repository: " + REPO, "Generated (UTC): " + NOW.isoformat(), "",
               "| Metric | Count |", "|---|---:|",
               "| Branches including default | " + str(len(branches)) + " |",
               "| Open PRs | " + str(len(prs)) + " |",
               "| Open issues | " + str(len(issues)) + " |",
               "| Branches without open PR (review only) | " + str(len(findings)) + " |",
               "| Stale PRs (>= " + str(STALE_DAYS) + " days) | " + str(len(stale_prs)) + " |",
               "| Stale issues | " + str(len(stale_issues)) + " |", "",
               "**Do not mark branches unfinished or delete them based on this report.**",
               "Consult the project's canonical ledger, closed/merged PRs and main code first.", "",
               "## Stale PRs"]
    summary.extend("- [#" + str(p["number"]) + "](" + p["url"] + ") (" + str(p["days"]) + "d): " + p["title"].replace("\n", " ") for p in stale_prs)
    summary.append("\n## Stale issues")
    summary.extend("- [#" + str(i["number"]) + "](" + i["url"] + ") (" + str(i["days"]) + "d): " + i["title"].replace("\n", " ") for i in stale_issues)
    summary.append("\nBranch candidates are available in report.json artifact.")
    text = "\n".join(summary) + "\n"
    with open("inventory-output/summary.md", "w", encoding="utf-8") as f:
        f.write(text)
    with open(os.environ.get("GITHUB_STEP_SUMMARY", "/dev/null"), "a", encoding="utf-8") as f:
        f.write(text)
    print(text)

if __name__ == "__main__":
    try:
        main()
    except (urllib.error.URLError, KeyError, ValueError, RuntimeError) as error:
        print("Inventory failed: " + str(error), file=sys.stderr)
        sys.exit(1)

"""Offline tests for the read-only inventory; no GitHub token or network required."""
import importlib.util
import json
import os
import pathlib
import tempfile
import unittest
from unittest.mock import patch

SCRIPT = pathlib.Path(__file__).resolve().parents[1] / "scripts" / "development_inventory.py"
os.environ.update(GITHUB_REPOSITORY="example/demo")
os.environ.setdefault("GITHUB_TOKEN", "offline-test-token")
spec = importlib.util.spec_from_file_location("development_inventory", SCRIPT)
inventory = importlib.util.module_from_spec(spec)
spec.loader.exec_module(inventory)

class InventoryTests(unittest.TestCase):
    def test_age_and_missing(self):
        self.assertIsNone(inventory.age(None))
        self.assertEqual(inventory.age(inventory.NOW.isoformat()), 0)

    def test_pagination(self):
        calls = []
        def fake_get(path):
            calls.append(path)
            return [{"name": str(i)} for i in range(100)] if path.rsplit("page=", 1)[-1] == "1" else [{"name": "last"}]
        with patch.object(inventory, "get", side_effect=fake_get):
            result = inventory.pages("/repos/example/demo/branches")
        self.assertEqual(len(result), 101)
        self.assertEqual(len(calls), 2)

    def test_pagination_multiple_full_pages(self):
        calls = []
        def fake_get(path):
            calls.append(path)
            page = int(path.rsplit("page=", 1)[-1])
            if page <= 2:
                return [{"name": "branch-%s-%s" % (page, i)} for i in range(100)]
            return [{"name": "branch-last"}]
        with patch.object(inventory, "get", side_effect=fake_get):
            result = inventory.pages("/repos/example/demo/pulls?state=all")
        self.assertEqual(len(result), 201)
        self.assertEqual(len(calls), 3)
        self.assertTrue(all("state=all" in path for path in calls))
        self.assertEqual(result[-1]["name"], "branch-last")

    def test_empty_page_terminates_pagination(self):
        with patch.object(inventory, "get", return_value=[]) as fake_get:
            result = inventory.pages("/repos/example/demo/issues?state=open")
        self.assertEqual(result, [])
        fake_get.assert_called_once_with("/repos/example/demo/issues?state=open&per_page=100&page=1")

    def test_pr_and_issue_pagination_are_independent(self):
        calls = []
        def fake_get(path):
            calls.append(path)
            page = int(path.rsplit("page=", 1)[-1])
            if "/pulls?" in path:
                return [{"number": i} for i in range(100)] if page == 1 else [{"number": 101}]
            if "/issues?" in path:
                return [{"number": i} for i in range(100)] if page == 1 else [{"number": 102}]
            raise AssertionError(path)
        with patch.object(inventory, "get", side_effect=fake_get):
            prs = inventory.pages("/repos/example/demo/pulls?state=all")
            issues = inventory.pages("/repos/example/demo/issues?state=open")
        self.assertEqual(len(prs), 101)
        self.assertEqual(len(issues), 101)
        self.assertEqual(len(calls), 4)

    def test_closed_pr_history_is_retained(self):
        old = "2020-01-01T00:00:00Z"
        closed = {"number": 8, "state": "closed", "title": "Done",
                  "updated_at": old, "html_url": "https://github.com/example/demo/pull/8",
                  "head": {"ref": "feature/done", "repo": {"full_name": "example/demo"}}}
        def fake_pages(path):
            if "/branches" in path:
                return [{"name": "main"}, {"name": "feature/done"}]
            if "/pulls?" in path:
                return [closed]
            if "/issues?" in path:
                return []
            raise AssertionError(path)
        with tempfile.TemporaryDirectory() as tmp:
            current = os.getcwd()
            try:
                os.chdir(tmp)
                with patch.object(inventory, "get", return_value={"default_branch": "main"}), patch.object(inventory, "pages", side_effect=fake_pages):
                    inventory.main()
                with open("inventory-output/report.json", encoding="utf-8") as f:
                    result = json.load(f)
            finally:
                os.chdir(current)
        self.assertEqual(result["open_pr_count"], 0)
        self.assertEqual(result["historical_pr_count"], 1)
        self.assertTrue(result["review_candidates"][0]["historical_pr_found"])

    def test_report_candidates_are_not_completion_claims(self):
        old = "2020-01-01T00:00:00Z"
        recent = inventory.NOW.isoformat()
        pr = {"number": 12, "state": "open", "title": "Open PR", "updated_at": old,
              "html_url": "https://github.com/example/demo/pull/12",
              "head": {"ref": "feature/open", "repo": {"full_name": "example/demo"}}}
        issue = {"number": 13, "title": "Open issue", "updated_at": recent,
                 "html_url": "https://github.com/example/demo/issues/13"}
        def fake_pages(path):
            if "/branches" in path:
                return [{"name": "main"}, {"name": "feature/open"}, {"name": "research/archive"}]
            if "/pulls?" in path:
                return [pr]
            if "/issues?" in path:
                return [issue, {"pull_request": {}, **issue}]
            raise AssertionError(path)
        with tempfile.TemporaryDirectory() as tmp:
            current = os.getcwd()
            try:
                os.chdir(tmp)
                with patch.object(inventory, "get", return_value={"default_branch": "main"}), patch.object(inventory, "pages", side_effect=fake_pages):
                    inventory.main()
                with open("inventory-output/report.json", encoding="utf-8") as f:
                    result = json.load(f)
            finally:
                os.chdir(current)
        self.assertEqual(result["branch_count"], 3)
        self.assertEqual(result["open_issue_count"], 1)
        self.assertEqual([x["name"] for x in result["review_candidates"]], ["research/archive"])
        self.assertEqual(len(result["stale_prs"]), 1)
        self.assertEqual(len(result["stale_issues"]), 0)
        self.assertIn("NOT proof", result["caveat"])

if __name__ == "__main__":
    unittest.main()
